### 1.2 Mock MCP Server

Create a lightweight mock server that simulates unifi-mcp responses for isolated testing.

**File: `tests/claude-code/mocks/mock-mcp-server.py`**

```python
#!/usr/bin/env python3
"""
Mock MCP Server for Claude Code Integration Testing

This server simulates unifi-mcp responses without requiring
actual UniFi hardware. Used for testing skills, commands, and hooks.

Usage:
    python mock-mcp-server.py [--port 8765]
"""

import json
import sys
from pathlib import Path

# Mock data storage
MOCK_DATA = {
    "clients": [],
    "networks": [],
    "devices": [],
    "firewall_rules": [],
    "change_log": []
}

def load_fixtures():
    """Load test fixtures from JSON files."""
    fixtures_dir = Path(__file__).parent.parent / "fixtures"

    if (fixtures_dir / "sample-clients.json").exists():
        with open(fixtures_dir / "sample-clients.json") as f:
            MOCK_DATA["clients"] = json.load(f)

    if (fixtures_dir / "sample-networks.json").exists():
        with open(fixtures_dir / "sample-networks.json") as f:
            MOCK_DATA["networks"] = json.load(f)

    if (fixtures_dir / "sample-devices.json").exists():
        with open(fixtures_dir / "sample-devices.json") as f:
            MOCK_DATA["devices"] = json.load(f)


# Tool handlers
def unifi_list_clients(params: dict) -> dict:
    """Mock implementation of unifi_list_clients."""
    clients = MOCK_DATA["clients"]

    # Filter by network if specified
    if "network" in params:
        clients = [c for c in clients if c.get("network") == params["network"]]

    return {"clients": clients, "count": len(clients)}


def unifi_find_client(params: dict) -> dict:
    """Mock implementation of unifi_find_client."""
    query = params.get("query", "").lower()

    for client in MOCK_DATA["clients"]:
        if query in client.get("name", "").lower():
            return {"found": True, "client": client}
        if query in client.get("mac", "").lower():
            return {"found": True, "client": client}
        if query == client.get("ip", ""):
            return {"found": True, "client": client}

    return {"found": False, "client": None}


def unifi_list_networks(params: dict) -> dict:
    """Mock implementation of unifi_list_networks."""
    return {"networks": MOCK_DATA["networks"]}


def unifi_create_vlan(params: dict) -> dict:
    """Mock implementation of unifi_create_vlan."""
    vlan = {
        "id": f"net_{len(MOCK_DATA['networks']) + 1}",
        "name": params.get("name"),
        "vlan_id": params.get("vlan_id"),
        "subnet": params.get("subnet"),
        "purpose": params.get("purpose", "corporate")
    }
    MOCK_DATA["networks"].append(vlan)

    MOCK_DATA["change_log"].append({
        "action": "create_vlan",
        "params": params,
        "result": vlan
    })

    return {"success": True, "network": vlan}


def unifi_delete_network(params: dict) -> dict:
    """Mock implementation of unifi_delete_network."""
    network_id = params.get("network_id")

    for i, net in enumerate(MOCK_DATA["networks"]):
        if net["id"] == network_id:
            deleted = MOCK_DATA["networks"].pop(i)
            MOCK_DATA["change_log"].append({
                "action": "delete_network",
                "params": params,
                "result": deleted
            })
            return {"success": True, "deleted": deleted}

    return {"success": False, "error": "Network not found"}


def unifi_create_firewall_rule(params: dict) -> dict:
    """Mock implementation of unifi_create_firewall_rule."""
    rule = {
        "id": f"rule_{len(MOCK_DATA['firewall_rules']) + 1}",
        "name": params.get("name"),
        "action": params.get("action"),
        "source": params.get("source"),
        "destination": params.get("destination"),
        "protocol": params.get("protocol", "all"),
        "enabled": True
    }
    MOCK_DATA["firewall_rules"].append(rule)

    MOCK_DATA["change_log"].append({
        "action": "create_firewall_rule",
        "params": params,
        "result": rule
    })

    return {"success": True, "rule": rule}


def unifi_block_client(params: dict) -> dict:
    """Mock implementation of unifi_block_client."""
    mac = params.get("mac")

    for client in MOCK_DATA["clients"]:
        if client.get("mac") == mac:
            client["blocked"] = True
            MOCK_DATA["change_log"].append({
                "action": "block_client",
                "params": params,
                "result": client
            })
            return {"success": True, "client": client}

    return {"success": False, "error": "Client not found"}


def unifi_unblock_client(params: dict) -> dict:
    """Mock implementation of unifi_unblock_client."""
    mac = params.get("mac")

    for client in MOCK_DATA["clients"]:
        if client.get("mac") == mac:
            client["blocked"] = False
            MOCK_DATA["change_log"].append({
                "action": "unblock_client",
                "params": params,
                "result": client
            })
            return {"success": True, "client": client}

    return {"success": False, "error": "Client not found"}


def unifi_assign_client_network(params: dict) -> dict:
    """Mock implementation of unifi_assign_client_network."""
    mac = params.get("mac")
    network_id = params.get("network_id")

    for client in MOCK_DATA["clients"]:
        if client.get("mac") == mac:
            old_network = client.get("network")
            client["network"] = network_id
            MOCK_DATA["change_log"].append({
                "action": "assign_client_network",
                "params": params,
                "result": {"old": old_network, "new": network_id}
            })
            return {"success": True, "client": client}

    return {"success": False, "error": "Client not found"}


# Tool dispatch
TOOLS = {
    "unifi_list_clients": unifi_list_clients,
    "unifi_find_client": unifi_find_client,
    "unifi_list_networks": unifi_list_networks,
    "unifi_create_vlan": unifi_create_vlan,
    "unifi_delete_network": unifi_delete_network,
    "unifi_create_firewall_rule": unifi_create_firewall_rule,
    "unifi_block_client": unifi_block_client,
    "unifi_unblock_client": unifi_unblock_client,
    "unifi_assign_client_network": unifi_assign_client_network,
}


def handle_tool_call(tool_name: str, params: dict) -> dict:
    """Dispatch tool call to handler.

    Accepts both formats:
    - Short form: "unifi_list_clients"
    - MCP form: "mcp__unifi__unifi_list_clients"
    """
    # Normalize MCP-prefixed names to short form
    normalized_name = tool_name
    if tool_name.startswith("mcp__unifi__"):
        normalized_name = tool_name.replace("mcp__unifi__", "")

    if normalized_name in TOOLS:
        return TOOLS[normalized_name](params)
    return {"error": f"Unknown tool: {tool_name}"}


def get_change_log() -> list:
    """Return the change log for verification."""
    return MOCK_DATA["change_log"]


def reset_mock_data():
    """Reset mock data to initial state."""
    MOCK_DATA["change_log"] = []
    load_fixtures()


if __name__ == "__main__":
    load_fixtures()
    print("Mock MCP Server loaded with fixtures")
    print(f"Clients: {len(MOCK_DATA['clients'])}")
    print(f"Networks: {len(MOCK_DATA['networks'])}")
    print(f"Devices: {len(MOCK_DATA['devices'])}")
```

### 2.1 Hook Test Framework

**File: `tests/claude-code/hooks/run-hook-tests.sh`**

```bash
#!/bin/bash
#
# Hook Test Runner
# Executes all hook unit tests and reports results
#

set -e

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
HOOKS_DIR="$(dirname "$SCRIPT_DIR")/../.claude/hooks"
PASSED=0
FAILED=0
TESTS=()

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

log_pass() {
    echo -e "${GREEN}[PASS]${NC} $1"
    ((PASSED++))
}

log_fail() {
    echo -e "${RED}[FAIL]${NC} $1"
    ((FAILED++))
}

log_info() {
    echo -e "${YELLOW}[INFO]${NC} $1"
}

# Run individual test scripts
run_test() {
    local test_script="$1"
    local test_name=$(basename "$test_script" .sh)

    log_info "Running: $test_name"

    if bash "$test_script" "$HOOKS_DIR"; then
        log_pass "$test_name"
    else
        log_fail "$test_name"
    fi
}

echo "========================================"
echo "Claude Code Hook Test Suite"
echo "========================================"
echo ""

# Find and run all test scripts
for test_file in "$SCRIPT_DIR"/test-*.sh; do
    if [[ -f "$test_file" ]]; then
        run_test "$test_file"
    fi
done

echo ""
echo "========================================"
echo "Results: $PASSED passed, $FAILED failed"
echo "========================================"

if [[ $FAILED -gt 0 ]]; then
    exit 1
fi
```

#!/bin/bash
#
# Tests for post-change-log.sh hook
#

HOOKS_DIR="${1:-.claude/hooks}"
HOOK_SCRIPT="$HOOKS_DIR/post-change-log.sh"
TEMP_DIR=$(mktemp -d)
trap "rm -rf $TEMP_DIR" EXIT

# Setup test environment
export CLAUDE_CHANGE_LOG="$TEMP_DIR/changes.json"

# Test 1: Hook creates log file if not exists
test_creates_log_file() {
    rm -f "$CLAUDE_CHANGE_LOG"

    # Simulate tool call result (using correct tool_name/tool_input format)
    echo '{"tool_name": "mcp__unifi__unifi_create_vlan", "tool_input": {"name": "TestVLAN"}, "result": {"success": true}}' | \
        bash "$HOOK_SCRIPT"

    if [[ -f "$CLAUDE_CHANGE_LOG" ]]; then
        return 0
    else
        echo "Log file was not created"
        return 1
    fi
}

# Test 2: Hook appends entries correctly
test_appends_entries() {
    rm -f "$CLAUDE_CHANGE_LOG"

    # First entry
    echo '{"tool_name": "mcp__unifi__unifi_create_vlan", "tool_input": {"name": "VLAN1"}, "result": {"success": true}}' | \
        bash "$HOOK_SCRIPT"

    # Second entry
    echo '{"tool_name": "mcp__unifi__unifi_create_firewall_rule", "tool_input": {"name": "Rule1"}, "result": {"success": true}}' | \
        bash "$HOOK_SCRIPT"

    # Count entries (each should be on its own line in JSONL format)
    local count=$(wc -l < "$CLAUDE_CHANGE_LOG")

    if [[ $count -eq 2 ]]; then
        return 0
    else
        echo "Expected 2 entries, got $count"
        return 1
    fi
}

# Test 3: Entries include timestamp
test_includes_timestamp() {
    rm -f "$CLAUDE_CHANGE_LOG"

    echo '{"tool_name": "mcp__unifi__unifi_block_client", "tool_input": {"mac": "aa:bb:cc:dd:ee:ff"}, "result": {"success": true}}' | \
        bash "$HOOK_SCRIPT"

    if grep -q '"timestamp"' "$CLAUDE_CHANGE_LOG"; then
        return 0
    else
        echo "Entry missing timestamp field"
        return 1
    fi
}

# Run tests
TESTS_PASSED=0
TESTS_FAILED=0

for test_func in test_creates_log_file test_appends_entries test_includes_timestamp; do
    if $test_func; then
        ((TESTS_PASSED++))
    else
        ((TESTS_FAILED++))
    fi
done

echo "Hook tests: $TESTS_PASSED passed, $TESTS_FAILED failed"
[[ $TESTS_FAILED -eq 0 ]]

#!/bin/bash
#
# Tests for pre-delete-network.sh hook
#

HOOKS_DIR="${1:-.claude/hooks}"
HOOK_SCRIPT="$HOOKS_DIR/pre-delete-network.sh"

# Test 1: Blocks deletion of 'default' network
test_blocks_default_network() {
    local result=$(echo '{"tool_name": "mcp__unifi__unifi_delete_network", "tool_input": {"identifier": "default"}}' | bash "$HOOK_SCRIPT" 2>&1)
    local exit_code=$?

    if [[ $exit_code -ne 0 ]] && echo "$result" | grep -qi "protected"; then
        return 0
    else
        echo "Should block deletion of default network"
        return 1
    fi
}

# Test 2: Blocks deletion of 'management' network
test_blocks_management_network() {
    local result=$(echo '{"tool_name": "mcp__unifi__unifi_delete_network", "tool_input": {"identifier": "management"}}' | bash "$HOOK_SCRIPT" 2>&1)
    local exit_code=$?

    if [[ $exit_code -ne 0 ]]; then
        return 0
    else
        echo "Should block deletion of management network"
        return 1
    fi
}

# Test 3: Allows deletion of custom networks
test_allows_custom_network() {
    local result=$(echo '{"tool_name": "mcp__unifi__unifi_delete_network", "tool_input": {"identifier": "my_custom_vlan"}}' | bash "$HOOK_SCRIPT" 2>&1)
    local exit_code=$?

    if [[ $exit_code -eq 0 ]]; then
        return 0
    else
        echo "Should allow deletion of custom networks"
        return 1
    fi

test_blocks_name_variants() {

            return 1
        fi

    done

    return 0
}

# Run tests

for test_func in test_blocks_default_network test_blocks_management_network test_allows_custom_network test_blocks_name_variants; do

        ((TESTS_PASSED++))

        ((TESTS_FAILED++))
