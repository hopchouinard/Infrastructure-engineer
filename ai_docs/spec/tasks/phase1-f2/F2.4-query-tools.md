# F2.4: Query Tools

## Objective

Implement all read-only query tools for the UniFi MCP server, including tools for listing/getting VLANs, clients, devices, firewall rules, and port status.

**Architecture Note:** This task defines data models and tool implementation functions. The actual MCP tool registration happens in F2.9 using the FastMCP decorator pattern (`@mcp.tool()`). The functions defined here are called by the decorated tools in F2.9.

## Prerequisites

- F2.1 (Project Setup) completed
- F2.2 (Configuration) completed
- F2.3 (API Client) completed

## Scope

### In Scope

- Data models for networks, clients, devices, firewall rules, ports
- Query tool implementation functions
- Response data transformation
- Integration with F2.9's FastMCP tools

### Out of Scope

- Write operations (F2.5)
- Caching
- Real-time updates
- MCP tool registration (F2.9)

from __future__ import annotations

from dataclasses import dataclass
from typing import Any


@dataclass
class SwitchPort:
    """Represents a switch port."""

    port_idx: int
    name: str | None
    enabled: bool
    speed: int  # Mbps
    full_duplex: bool
    is_uplink: bool
    poe_mode: str | None  # off, auto, pasv24, passthrough
    poe_power: float | None  # Watts
    native_vlan: int | None
    tagged_vlans: list[int]
    port_profile_id: str | None
    # Stats
    rx_bytes: int
    tx_bytes: int
    rx_packets: int
    tx_packets: int

    @property
    def status(self) -> str:
        """Get port status."""
        if not self.enabled:
            return "disabled"
        if self.speed == 0:
            return "down"
        return "up"

    @property
    def speed_display(self) -> str:
        """Get human-readable speed."""
        if self.speed == 0:
            return "N/A"
        if self.speed >= 1000:
            return f"{self.speed // 1000}G"
        return f"{self.speed}M"

    @classmethod
    def from_api_response(cls, data: dict[str, Any]) -> "SwitchPort":
        """Create SwitchPort from API response data."""
        return cls(
            port_idx=data.get("port_idx", 0),
            name=data.get("name"),
            enabled=data.get("enable", True),
            speed=data.get("speed", 0),
            full_duplex=data.get("full_duplex", True),
            is_uplink=data.get("is_uplink", False),
            poe_mode=data.get("poe_mode"),
            poe_power=data.get("poe_power"),
            native_vlan=data.get("native_networkconf_id"),
            tagged_vlans=data.get("tagged_networkconf_ids", []),
            port_profile_id=data.get("portconf_id"),
            rx_bytes=data.get("rx_bytes", 0),
            tx_bytes=data.get("tx_bytes", 0),
            rx_packets=data.get("rx_packets", 0),
            tx_packets=data.get("tx_packets", 0),
        )

    def to_dict(self) -> dict[str, Any]:
        """Convert to dictionary for tool response."""
        result: dict[str, Any] = {
            "port": self.port_idx,
            "name": self.name,
            "status": self.status,
            "enabled": self.enabled,
            "speed": self.speed_display,
            "is_uplink": self.is_uplink,
        }

        if self.poe_mode and self.poe_mode != "off":
            result["poe"] = {
                "mode": self.poe_mode,
                "power_watts": self.poe_power,
            }

        if self.native_vlan:
            result["native_vlan"] = self.native_vlan
        if self.tagged_vlans:
            result["tagged_vlans"] = self.tagged_vlans

        result["stats"] = {
            "rx_bytes": self.rx_bytes,
            "tx_bytes": self.tx_bytes,
        }

        return result

"""
Network/VLAN query functions.

These functions are called by the FastMCP-decorated tools in server.py.
"""

from __future__ import annotations

from typing import Any

from unifi_mcp.client import UniFiClient
from unifi_mcp.models.network import Network


async def list_networks(
    client: UniFiClient,
    include_default: bool = True,
    vlan_only: bool = False,
) -> dict[str, Any]:
    """
    List all networks/VLANs configured on the controller.

    Args:
        client: UniFi API client
        include_default: Include default/management network
        vlan_only: Only show networks with VLAN IDs

    Returns:
        Dict with success status and network data
    """
    try:
        response = await client.get("/rest/networkconf")
        raw_networks = response.get("data", [])

        networks = []
        for data in raw_networks:
            network = Network.from_api_response(data)

            # Apply filters
            if vlan_only and network.vlan_id is None:
                continue
            if not include_default:
                if network.purpose == "wan" or network.name.lower() == "default":
                    continue

            networks.append(network.to_dict())

        return {
            "success": True,
            "data": networks,
            "message": f"Found {len(networks)} networks",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}


async def get_network(
    client: UniFiClient,
    identifier: str,
) -> dict[str, Any]:
    """
    Get detailed information about a specific network.

    Args:
        client: UniFi API client
        identifier: Network name, ID, or VLAN ID

    Returns:
        Dict with success status and network data
    """
    identifier_lower = identifier.lower()

    try:
        response = await client.get("/rest/networkconf")

        for data in response.get("data", []):
            network = Network.from_api_response(data)

            # Match by name, ID, or VLAN ID
            if (
                network.name.lower() == identifier_lower
                or network.id == identifier
                or str(network.vlan_id) == identifier_lower
            ):
                return {
                    "success": True,
                    "data": network.to_dict(),
                }

        return {
            "success": False,
            "error": f"Network not found: {identifier}",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}

from __future__ import annotations

from typing import Any

from unifi_mcp.client import UniFiClient
from unifi_mcp.models.client import Client


async def list_clients(
    client: UniFiClient,
    network: str | None = None,
    wired_only: bool = False,
    wireless_only: bool = False,
    include_blocked: bool = False,
) -> dict[str, Any]:
    """
    List all connected clients.

    Args:
        client: UniFi API client
        network: Filter by network/VLAN name
        wired_only: Only show wired clients
        wireless_only: Only show wireless clients
        include_blocked: Include blocked clients

    Returns:
        Dict with success status and client data
    """
    try:
        response = await client.get("/stat/sta")
        raw_clients = response.get("data", [])

        clients = []
        for data in raw_clients:
            cli = Client.from_api_response(data)

            # Apply filters
            if network:
                if cli.network and cli.network.lower() != network.lower():
                    continue
            if wired_only and not cli.is_wired:
                continue
            if wireless_only and cli.is_wired:
                continue
            if not include_blocked and cli.is_blocked:
                continue

            clients.append(cli.to_dict())

        return {
            "success": True,
            "data": clients,
            "message": f"Found {len(clients)} clients",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}


async def get_client(
    client: UniFiClient,
    identifier: str,
) -> dict[str, Any]:
    """
    Get detailed information about a specific client.

    Args:
        client: UniFi API client
        identifier: Client MAC address, IP address, or name

    Returns:
        Dict with success status and client data
    """
    identifier = identifier.lower().strip()

    try:
        response = await client.get("/stat/sta")

        for data in response.get("data", []):
            cli = Client.from_api_response(data)

            # Match by MAC, IP, name, or hostname
            if (
                cli.mac.lower() == identifier
                or (cli.ip and cli.ip.lower() == identifier)
                or (cli.name and cli.name.lower() == identifier)
                or (cli.hostname and cli.hostname.lower() == identifier)
            ):
                return {
                    "success": True,
                    "data": cli.to_dict(),
                }

        return {
            "success": False,
            "error": f"Client not found: {identifier}",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}


async def search_clients(
    client: UniFiClient,
    query: str,
) -> dict[str, Any]:
    """
    Search for clients by partial name or manufacturer.

    Args:
        client: UniFi API client
        query: Search query (partial match)

    Returns:
        Dict with success status and matching clients
    """
    query_lower = query.lower()

    try:
        response = await client.get("/stat/sta")
        matches = []

        for data in response.get("data", []):
            cli = Client.from_api_response(data)

            # Search in multiple fields
            searchable = " ".join(filter(None, [
                cli.name,
                cli.hostname,
                cli.manufacturer,
                cli.mac,
                cli.ip,
            ])).lower()

            if query_lower in searchable:
                matches.append(cli.to_dict())

        return {
            "success": True,
            "data": matches,
            "message": f"Found {len(matches)} matching clients",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}

    """
    List all firewall rules.

    Args:
        client: UniFi API client
        ruleset: Filter by ruleset
        enabled_only: Only show enabled rules

    Returns:
        Dict with success status and firewall rules
    """
    try:
        response = await client.get("/rest/firewallrule")
        raw_rules = response.get("data", [])

        rules = []
        for data in raw_rules:
            rule = FirewallRule.from_api_response(data)

            # Apply filters
            if ruleset != "all" and rule.ruleset != ruleset:
                continue
            if enabled_only and not rule.enabled:
                continue

            rules.append(rule.to_dict())

        # Sort by ruleset then index
        rules.sort(key=lambda r: (r["ruleset"], r["index"]))

        return {
            "success": True,
            "data": rules,
            "message": f"Found {len(rules)} firewall rules",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}


async def get_firewall_rule(
    client: UniFiClient,
    identifier: str,
) -> dict[str, Any]:
    """
    Get details of a specific firewall rule.

    Args:
        client: UniFi API client
        identifier: Rule name or ID

    Returns:
        Dict with success status and rule data
    """
    try:
        response = await client.get("/rest/firewallrule")

        for data in response.get("data", []):
            rule = FirewallRule.from_api_response(data)

            if rule.name.lower() == identifier.lower() or rule.id == identifier:
                return {
                    "success": True,
                    "data": rule.to_dict(),
                }

        return {
            "success": False,
            "error": f"Firewall rule not found: {identifier}",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}

"""
Switch port query functions.

These functions are called by the FastMCP-decorated tools in server.py.
"""

from __future__ import annotations

from typing import Any

from unifi_mcp.client import UniFiClient
from unifi_mcp.models.port import SwitchPort
from unifi_mcp.models.client import Client


async def get_port_status(
    client: UniFiClient,
    switch: str,
    port: int | None = None,
) -> dict[str, Any]:
    """
    Get status of switch ports.

    Args:
        client: UniFi API client
        switch: Switch name, MAC, or IP address
        port: Specific port number (optional)

    Returns:
        Dict with success status and port data
    """
    switch_id = switch.lower()

    try:
        response = await client.get("/stat/device")

        for data in response.get("data", []):
            # Check if this is the target switch
            if data.get("type") != "usw":
                continue

            name = (data.get("name") or "").lower()
            mac = (data.get("mac") or "").lower()
            ip = (data.get("ip") or "").lower()

            if switch_id not in [name, mac, ip]:
                continue

            # Found the switch, get port info
            port_table = data.get("port_table", [])
            ports = []

            for port_data in port_table:
                p = SwitchPort.from_api_response(port_data)

                if port is not None and p.port_idx != port:
                    continue

                ports.append(p.to_dict())

            if port is not None and not ports:
                return {
                    "success": False,
                    "error": f"Port {port} not found on switch",
                }

            return {
                "success": True,
                "data": ports if len(ports) != 1 else ports[0],
                "message": f"Found {len(ports)} port(s)",
            }

        return {
            "success": False,
            "error": f"Switch not found: {switch}",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}


async def get_client_on_port(
    client: UniFiClient,
    switch: str,
    port: int,
) -> dict[str, Any]:
    """
    Find which client is connected to a specific port.

    Args:
        client: UniFi API client
        switch: Switch name, MAC, or IP address
        port: Port number

    Returns:
        Dict with success status and client data (or None if empty)
    """
    switch_id = switch.lower()

    try:
        # First find the switch MAC
        devices = await client.get("/stat/device")
        switch_mac = None

        for device in devices.get("data", []):
            if device.get("type") != "usw":
                continue

            name = (device.get("name") or "").lower()
            mac = (device.get("mac") or "").lower()
            ip = (device.get("ip") or "").lower()

            if switch_id in [name, mac, ip]:
                switch_mac = device.get("mac", "").lower()
                break

        if not switch_mac:
            return {
                "success": False,
                "error": f"Switch not found: {switch}",
            }

        # Now find client on that port
        clients_response = await client.get("/stat/sta")

        for client_data in clients_response.get("data", []):
            cli = Client.from_api_response(client_data)

            if (
                cli.switch_mac
                and cli.switch_mac.lower() == switch_mac
                and cli.switch_port == port
            ):
                return {
                    "success": True,
                    "data": cli.to_dict(),
                }

        return {
            "success": True,
            "data": None,
            "message": f"No client connected to port {port}",
        }

    except Exception as e:
        return {"success": False, "error": str(e)}
