# F2.5: Configuration Tools

## Objective

Implement write operations for the UniFi MCP server, including tools for creating, updating, and deleting VLANs, firewall rules, and managing client/port configurations.

**Architecture Note:** This task defines configuration functions that are called by FastMCP-decorated tools registered in F2.9. Functions receive the UniFiClient, config, and rollback manager from the lifespan context state. Permission tiers are applied in F2.9 during tool registration.

## Prerequisites

- F2.4 (Query Tools) completed
- F2.6 (Safety System) - provides validation and rate limiting
- F2.7 (Rollback System) - provides rollback recording

## Scope

### In Scope

- VLAN/Network create/update/delete functions
- Firewall rule create/update/delete functions
- Switch port configuration functions
- Client management functions (block, unblock, rename)
- Dry-run mode support
- Rollback integration

### Out of Scope

- Bulk operations
- Device firmware updates
- Factory resets
- MCP tool registration (F2.9)

"""
Network/VLAN configuration functions.

These functions are called by the FastMCP-decorated tools in server.py.
They receive the UniFiClient and optional rollback manager from the lifespan context.
"""

from __future__ import annotations

import ipaddress
from typing import Any, TYPE_CHECKING

from unifi_mcp.client import UniFiClient

if TYPE_CHECKING:
    from unifi_mcp.safety.rollback import RollbackManager


async def validate_create_network(
    client: UniFiClient,
    name: str,
    vlan_id: int,
    subnet: str,
) -> list[str]:
    """
    Validate network creation parameters.

    Args:
        client: UniFi API client
        name: Network name
        vlan_id: VLAN ID (2-4094)
        subnet: CIDR notation

    Returns:
        List of validation errors (empty if valid)
    """
    errors = []

    # Validate subnet format
    try:
        ipaddress.ip_network(subnet, strict=False)
    except ValueError as e:
        errors.append(f"Invalid subnet: {e}")
        return errors

    # Check VLAN ID and name not in use
    response = await client.get("/rest/networkconf")
    existing = response.get("data", [])

    for net in existing:
        if net.get("vlan") == vlan_id:
            errors.append(f"VLAN ID {vlan_id} already in use")
        if net.get("name", "").lower() == name.lower():
            errors.append(f"Network name '{name}' already exists")

    return errors


async def create_network(
    client: UniFiClient,
    name: str,
    vlan_id: int,
    subnet: str,
    purpose: str = "corporate",
    dhcp_enabled: bool = True,
    dhcp_start: str | None = None,
    dhcp_stop: str | None = None,
    domain_name: str | None = None,
    dry_run: bool = False,
    rollback_manager: "RollbackManager | None" = None,
) -> dict[str, Any]:
    """
    Create a new VLAN/network.

    Args:
        client: UniFi API client
        name: Network name (1-32 chars)
        vlan_id: VLAN ID (2-4094)
        subnet: CIDR notation (e.g., 192.168.10.0/24)
        purpose: Network purpose ("corporate", "guest", "vlan-only")
        dhcp_enabled: Enable DHCP server
        dhcp_start: DHCP range start (auto-calculated if not specified)
        dhcp_stop: DHCP range end (auto-calculated if not specified)
        domain_name: Local domain name
        dry_run: Preview mode - don't actually make changes
        rollback_manager: Optional rollback manager for undo support

    Returns:
        Dict with success status and created network data
    """
    # Validate
    errors = await validate_create_network(client, name, vlan_id, subnet)
    if errors:
        return {"success": False, "error": "; ".join(errors)}

    # Check dry run
    if dry_run:
        return {
            "success": True,
            "message": f"[DRY RUN] Would create network '{name}' (VLAN {vlan_id})",
            "data": {"dry_run": True, "name": name, "vlan_id": vlan_id, "subnet": subnet},
        }

    # Parse subnet for DHCP config
    network = ipaddress.ip_network(subnet, strict=False)
    hosts = list(network.hosts())

    config = {
        "name": name,
        "vlan": vlan_id,
        "vlan_enabled": True,
        "ip_subnet": subnet,
        "purpose": purpose,
        "dhcpd_enabled": dhcp_enabled,
        "networkgroup": "LAN",
    }

    if dhcp_enabled and len(hosts) > 10:
        config["dhcpd_start"] = dhcp_start or str(hosts[10])
        config["dhcpd_stop"] = dhcp_stop or str(hosts[-2])

    if domain_name:
        config["domain_name"] = domain_name

    try:
        result = await client.post("/rest/networkconf", config)

        if result.get("meta", {}).get("rc") == "ok":
            network_data = result["data"][0]
            network_id = network_data["_id"]

            # Record for rollback
            rollback_id = None
            if rollback_manager:
                rollback_id = await rollback_manager.record(
                    action="create_network",
                    undo_action="delete_network",
                    undo_params={"network_id": network_id},
                    description=f"Created network '{name}'",
                )

            return {
                "success": True,
                "data": {"id": network_id, "name": name, "vlan_id": vlan_id},
                "message": f"Created network '{name}' (VLAN {vlan_id})",
                "rollback_id": rollback_id,
            }

        return {"success": False, "error": "API returned error"}

    except Exception as e:
        return {"success": False, "error": str(e)}
