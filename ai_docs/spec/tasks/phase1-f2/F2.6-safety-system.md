"""
Pre-execution validation for all tools.

Note: With the FastMCP pattern, validation is integrated into the tool registration
in server.py (F2.9). This module provides the ValidationEngine used by those tools.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass, field
from typing import TYPE_CHECKING, Any

from unifi_mcp.safety.tiers import PermissionTier, get_tier_behavior

if TYPE_CHECKING:
    from unifi_mcp.config import UniFiConfig

logger = logging.getLogger(__name__)


@dataclass
class ValidationResult:
    """Result of pre-execution validation."""

    valid: bool
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    requires_confirmation: bool = False
    confirmation_message: str | None = None


class ValidationEngine:
    """
    Pre-execution validation for all tools.

    Validates:
    - Tool-specific rules
    - Permission tier requirements
    - Rate limits
    - Dangerous operation detection
    """

    def __init__(self, config: "UniFiConfig") -> None:
        self.config = config
        self._write_count: int = 0
        self._last_reset: float = 0

    async def validate(
        self,
        tool_name: str,
        permission_tier: PermissionTier,
        params: dict[str, Any],
        tool_errors: list[str] | None = None,
    ) -> ValidationResult:
        """
        Run all validations before executing a tool.

        With FastMCP, tool-specific validation is done in the tool function itself.
        This method handles permission tier and rate limit validation.

        Args:
            tool_name: Name of the tool being executed
            permission_tier: Permission tier for the operation
            params: Tool parameters
            tool_errors: Errors from tool-specific validation (optional)

        Returns:
            ValidationResult with errors/warnings
        """
        errors: list[str] = tool_errors or []
        warnings: list[str] = []

        # Check rate limits for write operations
        tier_behavior = get_tier_behavior(permission_tier)
        if permission_tier != PermissionTier.READ:
            rate_error = await self._check_rate_limit()
            if rate_error:
                errors.append(rate_error)

        # Check if confirmation required
        requires_confirmation = tier_behavior.requires_confirmation
        confirmation_msg = None

        if requires_confirmation and not errors:
            confirmation_msg = self._generate_confirmation(
                tool_name, permission_tier, params
            )

        # Add warnings for high-risk operations
        if permission_tier in [PermissionTier.HIGH, PermissionTier.CRITICAL]:
            warnings.append(f"This is a {permission_tier.value}-risk operation")
            if self.config.enable_rollback:
                warnings.append("Rollback data will be saved")

        return ValidationResult(
            valid=len(errors) == 0,
            errors=errors,
            warnings=warnings,
            requires_confirmation=requires_confirmation,
            confirmation_message=confirmation_msg,
        )

    async def _check_rate_limit(self) -> str | None:
        """Check if write rate limit exceeded."""
        import time

        now = time.time()

        # Reset counter every minute
        if now - self._last_reset > 60:
            self._write_count = 0
            self._last_reset = now

        if self._write_count >= self.config.max_changes_per_minute:
            return (
                f"Rate limit exceeded: {self.config.max_changes_per_minute} "
                "changes per minute"
            )

        self._write_count += 1
        return None

    def _generate_confirmation(
        self,
        tool_name: str,
        permission_tier: PermissionTier,
        params: dict[str, Any],
    ) -> str:
        """Generate confirmation message for user."""
        tier_behavior = get_tier_behavior(permission_tier)

        if tier_behavior.confirmation_type == "brief":
            return f"Confirm: {tool_name}"

        elif tier_behavior.confirmation_type == "detailed":
            param_str = ", ".join(f"{k}={v}" for k, v in params.items())
            return f"Confirm: {tool_name}\nParameters: {param_str}"

        elif tier_behavior.confirmation_type == "explicit":
            lines = [
                f"⚠️ {permission_tier.value.upper()} RISK OPERATION",
                f"Action: {tool_name}",
                f"Parameters: {params}",
            ]
            if tier_behavior.show_rollback_info:
                lines.append("Rollback: Available for 60 minutes after execution")
            return "\n".join(lines)

        return f"Confirm: {tool.name}"


class DangerousOperationChecker:
    """Detect potentially dangerous operations."""

    DANGEROUS_PATTERNS = [
        # Network patterns
        ("delete", "network", "Deleting network will disconnect all clients"),
        ("delete", "vlan", "Deleting VLAN will disconnect all clients"),
        # Firewall patterns
        ("accept", "any", "Allowing all traffic is dangerous"),
        ("delete", "firewall", "Removing firewall rules may expose network"),
        # Client patterns
        ("block", "client", "Client will lose network access"),
    ]

    @classmethod
    def check(cls, tool_name: str, params: dict[str, Any]) -> list[str]:
        """Check for dangerous patterns in operation."""
        warnings = []
        tool_lower = tool_name.lower()
        params_str = str(params).lower()

        for pattern, keyword, warning in cls.DANGEROUS_PATTERNS:
            if pattern in tool_lower and keyword in params_str:
                warnings.append(warning)

        return warnings

```python
"""
Rate limiting for write operations.
"""

from __future__ import annotations

import asyncio
import time
from collections import deque
from dataclasses import dataclass


@dataclass
class RateLimitConfig:
    """Rate limit configuration."""
    max_requests: int = 5
    window_seconds: float = 60.0


class RateLimiter:
    """
    Token bucket rate limiter for write operations.

    Ensures we don't overwhelm the UniFi controller with changes.
    """

    def __init__(self, config: RateLimitConfig) -> None:
        self.config = config
        self._timestamps: deque[float] = deque()
        self._lock = asyncio.Lock()

    async def acquire(self) -> bool:
        """
        Attempt to acquire a rate limit token.

        Returns:
            True if allowed, False if rate limited
        """
        async with self._lock:
            now = time.monotonic()

            # Remove old timestamps
            while self._timestamps and self._timestamps[0] < now - self.config.window_seconds:
                self._timestamps.popleft()

            if len(self._timestamps) >= self.config.max_requests:
                return False

            self._timestamps.append(now)
            return True

    async def wait_and_acquire(self) -> None:
        """Wait until a token is available, then acquire."""
        while not await self.acquire():
            await asyncio.sleep(1.0)

    @property
    def available(self) -> int:
        """Number of tokens currently available."""
        now = time.monotonic()
        recent = sum(
            1 for t in self._timestamps
            if t >= now - self.config.window_seconds
        )
        return max(0, self.config.max_requests - recent)

    def reset(self) -> None:
        """Reset the rate limiter."""
        self._timestamps.clear()
```
