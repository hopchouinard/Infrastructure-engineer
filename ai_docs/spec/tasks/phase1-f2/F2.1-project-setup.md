# F2.1: Project Setup

## Objective

Set up the foundational project structure for the UniFi MCP Server, including directory layout, dependencies, development tooling, and base module files.

## Prerequisites

- Python 3.11+ available
- Understanding of MCP server architecture
- Feature 1 (Documentation Crawler) can inform API patterns (optional)

## Scope

### In Scope

- Directory structure creation
- pyproject.toml with dependencies
- Base module files with docstrings
- Development tooling configuration (ruff, mypy, pytest)
- .env.example file
- Basic error classes

### Out of Scope

- Actual implementation of tools
- API client implementation
- MCP server registration

"""
Custom exceptions for the UniFi MCP server.

Exception Hierarchy
------------------
UniFiMCPError (base)
├── AuthenticationError
├── ValidationError
├── APIError
├── RateLimitError
└── RollbackError
"""

from __future__ import annotations


class UniFiMCPError(Exception):
    """Base exception for all UniFi MCP server errors."""

    def __init__(self, message: str, details: dict | None = None) -> None:
        super().__init__(message)
        self.message = message
        self.details = details or {}

    def __str__(self) -> str:
        if self.details:
            return f"{self.message} - {self.details}"
        return self.message


class AuthenticationError(UniFiMCPError):
    """Failed to authenticate with UniFi controller."""

    pass


class ValidationError(UniFiMCPError):
    """Pre-execution validation failed."""

    def __init__(self, message: str, errors: list[str] | None = None) -> None:
        super().__init__(message, {"errors": errors or []})
        self.errors = errors or []


class APIError(UniFiMCPError):
    """UniFi API returned an error."""

    def __init__(
        self,
        message: str,
        status_code: int | None = None,
        response_body: dict | None = None,
    ) -> None:
        super().__init__(message, {
            "status_code": status_code,
            "response": response_body,
        })
        self.status_code = status_code
        self.response_body = response_body


class RateLimitError(UniFiMCPError):
    """Rate limit for write operations exceeded."""

    def __init__(self, message: str, retry_after: int | None = None) -> None:
        super().__init__(message, {"retry_after": retry_after})
        self.retry_after = retry_after


class RollbackError(UniFiMCPError):
    """Failed to execute rollback operation."""

    def __init__(self, message: str, rollback_id: str | None = None) -> None:
        super().__init__(message, {"rollback_id": rollback_id})
        self.rollback_id = rollback_id

"""
Helper functions for MCP tools.

Provides common utilities used across tool implementations.
"""

from __future__ import annotations

import json
from typing import Any


def format_success_response(
    data: dict[str, Any] | list[Any],
    message: str | None = None,
    rollback_id: str | None = None,
) -> str:
    """
    Format a successful tool response as JSON string.

    Args:
        data: Result data
        message: Optional success message
        rollback_id: Optional rollback ID for undo operations

    Returns:
        JSON string for MCP response
    """
    result: dict[str, Any] = {"success": True, "data": data}
    if message:
        result["message"] = message
    if rollback_id:
        result["rollback_id"] = rollback_id
    return json.dumps(result, indent=2)


def format_error_response(error: str, details: dict[str, Any] | None = None) -> str:
    """
    Format an error tool response as JSON string.

    Args:
        error: Error message
        details: Optional error details

    Returns:
        JSON string for MCP response
    """
    result: dict[str, Any] = {"success": False, "error": error}
    if details:
        result["details"] = details
    return json.dumps(result, indent=2)
```

"""
Permission tier definitions and behaviors.
"""

from __future__ import annotations

from dataclasses import dataclass
from enum import Enum
from typing import Literal


class PermissionTier(Enum):
    """Permission levels for tool operations."""

    READ = "read"
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    CRITICAL = "critical"


@dataclass(frozen=True)
class TierBehavior:
    """Behavior configuration for a permission tier."""

    requires_confirmation: bool
    confirmation_type: Literal["none", "brief", "detailed", "explicit"] = "none"
    show_rollback_info: bool = False
    user_initiated_only: bool = False
    log_level: Literal["DEBUG", "INFO", "WARNING", "CRITICAL"] = "INFO"


TIER_BEHAVIORS: dict[PermissionTier, TierBehavior] = {
    PermissionTier.READ: TierBehavior(
        requires_confirmation=False,
        confirmation_type="none",
        log_level="DEBUG",
    ),
    PermissionTier.LOW: TierBehavior(
        requires_confirmation=True,
        confirmation_type="brief",
        log_level="INFO",
    ),
    PermissionTier.MEDIUM: TierBehavior(
        requires_confirmation=True,
        confirmation_type="detailed",
        log_level="INFO",
    ),
    PermissionTier.HIGH: TierBehavior(
        requires_confirmation=True,
        confirmation_type="explicit",
        show_rollback_info=True,
        log_level="WARNING",
    ),
    PermissionTier.CRITICAL: TierBehavior(
        requires_confirmation=True,
        confirmation_type="explicit",
        show_rollback_info=True,
        user_initiated_only=True,
        log_level="CRITICAL",
    ),
}


def get_tier_behavior(tier: PermissionTier) -> TierBehavior:
    """Get the behavior configuration for a permission tier."""
    return TIER_BEHAVIORS[tier]
```
