## Objective

Implement the rollback system that allows undoing configuration changes made through the MCP server within a configurable retention period.

"""
Rollback management for configuration changes.
"""

from __future__ import annotations

import logging
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from typing import TYPE_CHECKING, Any, Callable, Awaitable

if TYPE_CHECKING:
    from unifi_mcp.client import UniFiClient

logger = logging.getLogger(__name__)


@dataclass
class RollbackEntry:
    """A single rollback entry."""

    id: str
    timestamp: datetime
    action: str
    undo_action: str
    undo_params: dict[str, Any]
    description: str
    expires_at: datetime
    executed: bool = False

    @property
    def is_expired(self) -> bool:
        """Check if this entry has expired."""
        return datetime.utcnow() > self.expires_at

    @property
    def time_remaining(self) -> timedelta:
        """Time remaining before expiration."""
        remaining = self.expires_at - datetime.utcnow()
        return max(remaining, timedelta(0))

    def to_dict(self) -> dict[str, Any]:
        """Convert to dictionary."""
        return {
            "id": self.id,
            "action": self.action,
            "description": self.description,
            "timestamp": self.timestamp.isoformat(),
            "expires_at": self.expires_at.isoformat(),
            "time_remaining_seconds": int(self.time_remaining.total_seconds()),
            "executed": self.executed,
        }


class RollbackManager:
    """
    Manages rollback data for configuration changes.

    Usage:
        rollback = RollbackManager(retention_minutes=60)

        # Record a change
        entry_id = await rollback.record(
            action="create_network",
            undo_action="delete_network",
            undo_params={"network_id": "abc123"},
            description="Created IoT network",
        )

        # Later, undo if needed
        result = await rollback.undo(entry_id, client)

        # List available rollbacks
        entries = rollback.list_available()
    """

    def __init__(self, retention_minutes: int = 60) -> None:
        """
        Initialize rollback manager.

        Args:
            retention_minutes: How long to keep rollback data
        """
        self._retention = timedelta(minutes=retention_minutes)
        self._entries: dict[str, RollbackEntry] = {}
        self._undo_handlers: dict[str, Callable[..., Awaitable[Any]]] = {}

    async def record(
        self,
        action: str,
        undo_action: str,
        undo_params: dict[str, Any],
        description: str = "",
    ) -> str:
        """
        Record a change for potential rollback.

        Args:
            action: Action that was performed
            undo_action: Action to undo it
            undo_params: Parameters for undo operation
            description: Human-readable description

        Returns:
            Rollback entry ID
        """
        entry_id = str(uuid.uuid4())[:8]
        now = datetime.utcnow()

        self._entries[entry_id] = RollbackEntry(
            id=entry_id,
            timestamp=now,
            action=action,
            undo_action=undo_action,
            undo_params=undo_params,
            description=description or f"Undo {action}",
            expires_at=now + self._retention,
        )

        logger.info(f"Recorded rollback {entry_id}: {action}")
        self._cleanup_expired()

        return entry_id

    async def undo(
        self,
        entry_id: str,
        client: "UniFiClient",
    ) -> dict[str, Any]:
        """
        Execute rollback for a specific entry.

        Args:
            entry_id: Rollback entry ID
            client: UniFi API client

        Returns:
            Result of undo operation
        """
        entry = self._entries.get(entry_id)

        if not entry:
            return {"success": False, "error": f"Rollback entry not found: {entry_id}"}

        if entry.is_expired:
            return {"success": False, "error": f"Rollback entry has expired"}

        if entry.executed:
            return {"success": False, "error": "Rollback already executed"}

        logger.info(f"Executing rollback {entry_id}: {entry.undo_action}")

        try:
            result = await self._execute_undo(entry, client)
            entry.executed = True
            return {"success": True, "message": f"Rolled back: {entry.description}"}

        except Exception as e:
            logger.error(f"Rollback failed: {e}")
            return {"success": False, "error": str(e)}

    async def _execute_undo(
        self,
        entry: RollbackEntry,
        client: "UniFiClient",
    ) -> Any:
        """Execute the undo operation."""
        action = entry.undo_action
        params = entry.undo_params

        # Built-in undo handlers
        if action == "delete_network":
            return await client.delete(f"/rest/networkconf/{params['network_id']}")

        elif action == "create_network":
            # Recreate from stored config
            return await client.post("/rest/networkconf", params["config"])

        elif action == "restore_network":
            return await client.put(
                f"/rest/networkconf/{params['network_id']}",
                params["original"],
            )

        elif action == "delete_firewall_rule":
            return await client.delete(f"/rest/firewallrule/{params['rule_id']}")

        elif action == "create_firewall_rule":
            return await client.post("/rest/firewallrule", params["config"])

        elif action == "unblock_client":
            return await client.post("/cmd/stamgr", {
                "cmd": "unblock-sta",
                "mac": params["mac"],
            })

        elif action == "restore_port_config":
            return await client.put(
                f"/rest/device/{params['device_id']}",
                {"port_overrides": params["original_overrides"]},
            )

        else:
            raise ValueError(f"Unknown undo action: {action}")

    def list_available(self) -> list[dict[str, Any]]:
        """List all available (non-expired, non-executed) rollback entries."""
        self._cleanup_expired()
        return [
            entry.to_dict()
            for entry in self._entries.values()
            if not entry.is_expired and not entry.executed
        ]

    def get_entry(self, entry_id: str) -> RollbackEntry | None:
        """Get a specific rollback entry."""
        return self._entries.get(entry_id)

    def _cleanup_expired(self) -> None:
        """Remove expired entries."""
        expired = [
            eid for eid, entry in self._entries.items()
            if entry.is_expired
        ]
        for eid in expired:
            del self._entries[eid]
            logger.debug(f"Cleaned up expired rollback {eid}")


class UndoTool:
    """Tool for executing rollbacks."""

    name = "unifi_undo"
    description = "Undo a previous configuration change"
    permission_tier = "HIGH"

    def __init__(self, rollback_manager: RollbackManager) -> None:
        self.rollback = rollback_manager

    def get_schema(self) -> dict[str, Any]:
        return {
            "type": "object",
            "properties": {
                "rollback_id": {
                    "type": "string",
                    "description": "Rollback entry ID from previous operation",
                },
                "list": {
                    "type": "boolean",
                    "description": "List available rollbacks instead of executing",
                    "default": False,
                },
            },
            "required": [],
        }

    async def execute(
        self,
        params: dict[str, Any],
        client: "UniFiClient",
    ) -> dict[str, Any]:
        if params.get("list"):
            available = self.rollback.list_available()
            return {
                "success": True,
                "data": available,
                "message": f"Found {len(available)} available rollbacks",
            }

        if not params.get("rollback_id"):
            return {
                "success": False,
                "error": "rollback_id required (or use list=true)",
            }

        return await self.rollback.undo(params["rollback_id"], client)
