"""
Pytest configuration and shared fixtures.
"""

from __future__ import annotations

import json
import pytest
from pathlib import Path
from typing import Any

import respx
from httpx import Response

from unifi_mcp.config import UniFiConfig


@pytest.fixture
def test_config() -> UniFiConfig:
    """Create a test configuration."""
    return UniFiConfig(
        unifi_host="https://192.168.1.1",
        unifi_username="test-user",
        unifi_password="test-password",
        unifi_site="default",
        unifi_verify_ssl=False,
        dry_run=True,
        max_changes_per_minute=100,
        rollback_retention_minutes=5,
    )


@pytest.fixture
def fixtures_dir() -> Path:
    """Get the fixtures directory path."""
    return Path(__file__).parent / "fixtures"


@pytest.fixture
def api_responses(fixtures_dir: Path) -> dict[str, Any]:
    """Load all API response fixtures."""
    responses = {}
    api_dir = fixtures_dir / "api_responses"

    if api_dir.exists():
        for json_file in api_dir.glob("*.json"):
            with open(json_file) as f:
                responses[json_file.stem] = json.load(f)

    return responses


@pytest.fixture
def mock_unifi_auth():
    """Mock UniFi authentication."""
    respx.post("https://192.168.1.1/api/auth/login").mock(
        return_value=Response(
            200,
            json={"meta": {"rc": "ok"}},
            headers={"X-CSRF-Token": "test-csrf"},
            cookies={"TOKEN": "test-token"},
        )
    )
    respx.post("https://192.168.1.1/api/auth/logout").mock(
        return_value=Response(200, json={"meta": {"rc": "ok"}})
    )


@pytest.fixture
def mock_unifi_api(mock_unifi_auth, api_responses):
    """Mock common UniFi API endpoints."""
    base = "https://192.168.1.1/proxy/network/api/s/default"

    # Networks
    if "networks" in api_responses:
        respx.get(f"{base}/rest/networkconf").mock(
            return_value=Response(200, json=api_responses["networks"])
        )

    # Clients
    if "clients" in api_responses:
        respx.get(f"{base}/stat/sta").mock(
            return_value=Response(200, json=api_responses["clients"])
        )

    # Devices
    if "devices" in api_responses:
        respx.get(f"{base}/stat/device").mock(
            return_value=Response(200, json=api_responses["devices"])
        )


def pytest_configure(config: pytest.Config) -> None:
    """Register custom markers."""
    config.addinivalue_line(
        "markers",
        "integration: marks tests requiring live API access",
    )


def pytest_collection_modifyitems(
    config: pytest.Config,
    items: list[pytest.Item],
) -> None:
    """Skip integration tests unless explicitly enabled."""
    if not config.getoption("--run-integration", default=False):
        skip_integration = pytest.mark.skip(
            reason="Integration tests disabled (use --run-integration)"
        )
        for item in items:
            if "integration" in item.keywords:
                item.add_marker(skip_integration)


def pytest_addoption(parser: pytest.Parser) -> None:
    """Add custom command line options."""
    parser.addoption(
        "--run-integration",
        action="store_true",
        default=False,
        help="Run integration tests against live API",
    )
```

"""
Server integration tests.
"""

import pytest
import respx
from httpx import Response

from unifi_mcp.server import UniFiMCPServer
from unifi_mcp.config import UniFiConfig


class TestUniFiMCPServer:
    """Test MCP server integration."""

    @respx.mock
    @pytest.mark.asyncio
    async def test_server_initialization(self, test_config, mock_unifi_api):
        """Server should initialize with all components."""
        server = UniFiMCPServer(test_config)

        assert server._tools is not None
        assert len(server._tools) > 0
        assert server._validation is not None
        assert server._rollback is not None

    @respx.mock
    @pytest.mark.asyncio
    async def test_tool_execution_with_validation(
        self, test_config, mock_unifi_api
    ):
        """Tool execution should run validation."""
        server = UniFiMCPServer(test_config)

        # Execute a query tool
        async with server._managed_client():
            result = await server._execute_tool(
                "unifi_list_networks",
                {}
            )

        assert result is not None

    @respx.mock
    @pytest.mark.asyncio
    async def test_tool_execution_with_audit(
        self, test_config, mock_unifi_api, tmp_path
    ):
        """Tool execution should be audited."""
        test_config.audit_log_path = tmp_path / "audit.log"
        test_config.enable_audit_log = True

        server = UniFiMCPServer(test_config)

        async with server._managed_client():
            await server._execute_tool("unifi_list_networks", {})

        # Check audit log was written
        assert test_config.audit_log_path.exists()

"""
Live API integration tests.

These tests run against a real UniFi controller.
Run with: pytest --run-integration
"""

import os
import pytest

from unifi_mcp.config import UniFiConfig
from unifi_mcp.client import UniFiClient


pytestmark = pytest.mark.integration


@pytest.fixture
def live_config() -> UniFiConfig:
    """Configuration from environment for live tests."""
    if not os.getenv("UNIFI_HOST"):
        pytest.skip("UNIFI_HOST not set")

    return UniFiConfig()


class TestLiveUniFiAPI:
    """Test against real UniFi controller (read-only)."""

    @pytest.mark.asyncio
    async def test_connection(self, live_config):
        """Should connect to live controller."""
        async with UniFiClient(live_config) as client:
            assert await client.test_connection()

    @pytest.mark.asyncio
    async def test_list_networks(self, live_config):
        """Should list networks from live controller."""
        async with UniFiClient(live_config) as client:
            result = await client.get("/rest/networkconf")

            assert result.get("meta", {}).get("rc") == "ok"
            assert "data" in result

    @pytest.mark.asyncio
    async def test_list_clients(self, live_config):
        """Should list clients from live controller."""
        async with UniFiClient(live_config) as client:
            result = await client.get("/stat/sta")

            assert result.get("meta", {}).get("rc") == "ok"

    @pytest.mark.asyncio
    async def test_list_devices(self, live_config):
        """Should list devices from live controller."""
        async with UniFiClient(live_config) as client:
            result = await client.get("/stat/device")

            assert result.get("meta", {}).get("rc") == "ok"
```

    "$@"

echo "Coverage report: coverage_html/index.html"
