```python
"""
Audit logging for all tool executions.
"""

from __future__ import annotations

import json
import logging
from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any

logger = logging.getLogger(__name__)


@dataclass
class AuditLogEntry:
    """A single audit log entry."""

    timestamp: str
    tool: str
    permission_tier: str
    parameters: dict[str, Any]
    result: str  # "success" | "failure" | "validation_error"
    message: str | None = None
    error: str | None = None
    rollback_id: str | None = None
    duration_ms: int = 0
    dry_run: bool = False

    def to_json(self) -> str:
        """Convert to JSON string."""
        return json.dumps(asdict(self), separators=(",", ":"))

    @classmethod
    def from_json(cls, line: str) -> "AuditLogEntry":
        """Parse from JSON string."""
        data = json.loads(line)
        return cls(**data)


class AuditLogger:
    """
    Logs all tool executions for audit trail.

    Log format: JSON Lines (one JSON object per line)
    File naming: unifi-audit.log (with optional rotation)

    Usage:
        audit = AuditLogger(Path("logs/unifi-audit.log"))

        entry = AuditLogEntry(
            timestamp=datetime.utcnow().isoformat(),
            tool="unifi_create_network",
            permission_tier="medium",
            parameters={"name": "IoT", "vlan_id": 10},
            result="success",
            message="Created network",
            duration_ms=342,
        )

        await audit.log(entry)
    """

    def __init__(self, log_path: Path) -> None:
        """
        Initialize audit logger.

        Args:
            log_path: Path to audit log file
        """
        self.log_path = log_path
        self._ensure_directory()

    def _ensure_directory(self) -> None:
        """Ensure log directory exists."""
        self.log_path.parent.mkdir(parents=True, exist_ok=True)

    async def log(self, entry: AuditLogEntry) -> None:
        """
        Append entry to audit log.

        Args:
            entry: Audit log entry to write
        """
        try:
            line = entry.to_json()
            with open(self.log_path, "a", encoding="utf-8") as f:
                f.write(line + "\n")

            # Also log to Python logging
            log_level = self._get_log_level(entry)
            logger.log(
                log_level,
                f"AUDIT: {entry.tool} - {entry.result} "
                f"({entry.duration_ms}ms)",
            )

        except Exception as e:
            logger.error(f"Failed to write audit log: {e}")

    def _get_log_level(self, entry: AuditLogEntry) -> int:
        """Determine log level based on entry."""
        if entry.result == "failure":
            return logging.ERROR
        if entry.permission_tier in ["high", "critical"]:
            return logging.WARNING
        return logging.INFO

    def query(
        self,
        tool: str | None = None,
        result: str | None = None,
        since: datetime | None = None,
        limit: int = 100,
    ) -> list[AuditLogEntry]:
        """
        Query audit log entries.

        Args:
            tool: Filter by tool name
            result: Filter by result (success/failure)
            since: Only entries after this timestamp
            limit: Maximum entries to return

        Returns:
            List of matching entries (newest first)
        """
        entries: list[AuditLogEntry] = []

        if not self.log_path.exists():
            return entries

        with open(self.log_path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue

                try:
                    entry = AuditLogEntry.from_json(line)

                    # Apply filters
                    if tool and entry.tool != tool:
                        continue
                    if result and entry.result != result:
                        continue
                    if since:
                        entry_time = datetime.fromisoformat(entry.timestamp)
                        if entry_time < since:
                            continue

                    entries.append(entry)

                except json.JSONDecodeError:
                    logger.warning(f"Invalid audit log line: {line[:50]}...")

        # Return newest first, limited
        entries.reverse()
        return entries[:limit]

    def get_statistics(self) -> dict[str, Any]:
        """Get audit log statistics."""
        entries = self.query(limit=10000)

        stats = {
            "total_entries": len(entries),
            "by_result": {},
            "by_tool": {},
            "by_tier": {},
            "average_duration_ms": 0,
        }

        if not entries:
            return stats

        total_duration = 0
        for entry in entries:
            # By result
            stats["by_result"][entry.result] = (
                stats["by_result"].get(entry.result, 0) + 1
            )
            # By tool
            stats["by_tool"][entry.tool] = (
                stats["by_tool"].get(entry.tool, 0) + 1
            )
            # By tier
            stats["by_tier"][entry.permission_tier] = (
                stats["by_tier"].get(entry.permission_tier, 0) + 1
            )
            total_duration += entry.duration_ms

        stats["average_duration_ms"] = total_duration // len(entries)

        return stats


def create_audit_entry(
    tool_name: str,
    permission_tier: str,
    params: dict[str, Any],
    result: dict[str, Any],
    duration_ms: int,
    dry_run: bool = False,
) -> AuditLogEntry:
    """
    Create an audit log entry from tool execution.

    Args:
        tool_name: Name of the tool
        permission_tier: Tool's permission tier
        params: Tool parameters (will be sanitized)
        result: Tool result
        duration_ms: Execution duration
        dry_run: Whether this was a dry run

    Returns:
        AuditLogEntry ready for logging
    """
    # Sanitize parameters (remove sensitive data)
    safe_params = _sanitize_params(params)

    return AuditLogEntry(
        timestamp=datetime.utcnow().isoformat() + "Z",
        tool=tool_name,
        permission_tier=permission_tier,
        parameters=safe_params,
        result="success" if result.get("success") else "failure",
        message=result.get("message"),
        error=result.get("error"),
        rollback_id=result.get("rollback_id"),
        duration_ms=duration_ms,
        dry_run=dry_run,
    )


def _sanitize_params(params: dict[str, Any]) -> dict[str, Any]:
    """Remove sensitive data from parameters."""
    sensitive_keys = {"password", "secret", "token", "key", "credential"}
    sanitized = {}

    for key, value in params.items():
        if any(s in key.lower() for s in sensitive_keys):
            sanitized[key] = "***REDACTED***"
        elif isinstance(value, dict):
            sanitized[key] = _sanitize_params(value)
        else:
            sanitized[key] = value

    return sanitized
```
