### 1. Authentication Module

**File: `mcp-servers/unifi-mcp/src/unifi_mcp/client/auth.py`**

```python
"""
UniFi API authentication handling.

The UniFi controller uses cookie-based authentication:
1. POST credentials to /api/auth/login
2. Receive session cookie (TOKEN) and CSRF token
3. Include both in subsequent requests
4. Session expires after inactivity (typically 30 min)
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import TYPE_CHECKING

import httpx

from unifi_mcp.errors import AuthenticationError

if TYPE_CHECKING:
    from unifi_mcp.config import UniFiConfig

logger = logging.getLogger(__name__)


@dataclass
class AuthSession:
    """Represents an authenticated session."""

    token: str
    csrf_token: str
    created_at: datetime
    expires_at: datetime | None = None

    @property
    def is_expired(self) -> bool:
        """Check if session appears expired."""
        if self.expires_at:
            return datetime.utcnow() > self.expires_at
        # Default assumption: 30 minute session timeout
        return datetime.utcnow() > self.created_at + timedelta(minutes=25)


class UniFiAuthenticator:
    """
    Handles authentication with UniFi controller.

    Usage:
        auth = UniFiAuthenticator(config)
        session = await auth.login()
        headers = auth.get_auth_headers()

        # Later, if session expired
        session = await auth.refresh()
    """

    def __init__(self, config: UniFiConfig) -> None:
        """
        Initialize the authenticator.

        Args:
            config: Server configuration
        """
        self.config = config
        self._session: AuthSession | None = None
        self._http_client: httpx.AsyncClient | None = None

    @property
    def is_authenticated(self) -> bool:
        """Check if we have a valid session."""
        return self._session is not None and not self._session.is_expired

    @property
    def session(self) -> AuthSession:
        """Get current session (raises if not authenticated)."""
        if self._session is None:
            raise AuthenticationError("Not authenticated - call login() first")
        return self._session

    def set_http_client(self, client: httpx.AsyncClient) -> None:
        """Set the HTTP client to use for auth requests."""
        self._http_client = client

    async def login(self) -> AuthSession:
        """
        Authenticate with the UniFi controller.

        Returns:
            AuthSession with tokens

        Raises:
            AuthenticationError: If login fails
        """
        if self._http_client is None:
            raise RuntimeError("HTTP client not set - call set_http_client first")

        logger.debug(f"Attempting login to {self.config.unifi_host}")

        try:
            response = await self._http_client.post(
                f"{self.config.unifi_host}/api/auth/login",
                json={
                    "username": self.config.unifi_username,
                    "password": self.config.unifi_password.get_secret_value(),
                },
            )

            if response.status_code == 200:
                # Extract session token from cookies
                token = response.cookies.get("TOKEN")
                if not token:
                    raise AuthenticationError(
                        "Login successful but no session token received"
                    )

                # Extract CSRF token from headers or response
                csrf_token = response.headers.get("X-CSRF-Token", "")

                self._session = AuthSession(
                    token=token,
                    csrf_token=csrf_token,
                    created_at=datetime.utcnow(),
                )

                logger.info("Successfully authenticated with UniFi controller")
                return self._session

            elif response.status_code == 401:
                raise AuthenticationError("Invalid username or password")
            elif response.status_code == 403:
                raise AuthenticationError(
                    "Access forbidden - check user permissions"
                )
            else:
                raise AuthenticationError(
                    f"Login failed with status {response.status_code}: "
                    f"{response.text}"
                )

        except httpx.ConnectError as e:
            raise AuthenticationError(
                f"Cannot connect to UniFi controller at {self.config.unifi_host}: {e}"
            ) from e
        except httpx.TimeoutException as e:
            raise AuthenticationError(
                f"Connection timeout to {self.config.unifi_host}"
            ) from e

    async def logout(self) -> None:
        """
        End the session with the controller.

        This invalidates the session token on the server side.
        """
        if self._session is None or self._http_client is None:
            return

        try:
            await self._http_client.post(
                f"{self.config.unifi_host}/api/auth/logout",
                headers=self.get_auth_headers(),
            )
            logger.debug("Successfully logged out")
        except Exception as e:
            logger.warning(f"Error during logout (non-fatal): {e}")
        finally:
            self._session = None

    async def refresh(self) -> AuthSession:
        """
        Refresh the session by re-authenticating.

        Returns:
            New AuthSession

        Raises:
            AuthenticationError: If refresh fails
        """
        logger.debug("Refreshing authentication session")
        self._session = None
        return await self.login()

    def get_auth_headers(self) -> dict[str, str]:
        """
        Get headers required for authenticated requests.

        Returns:
            Dictionary of auth headers

        Raises:
            AuthenticationError: If not authenticated
        """
        if self._session is None:
            raise AuthenticationError("Not authenticated")

        headers = {
            "Cookie": f"TOKEN={self._session.token}",
        }

        if self._session.csrf_token:
            headers["X-CSRF-Token"] = self._session.csrf_token

        return headers

    def get_cookies(self) -> dict[str, str]:
        """
        Get cookies for authenticated requests.

        Returns:
            Dictionary of cookies
        """
        if self._session is None:
            return {}
        return {"TOKEN": self._session.token}
```

```python
"""
Session management for UniFi API client.

Handles HTTP client lifecycle, connection pooling, and session state.
"""

from __future__ import annotations

import logging
from contextlib import asynccontextmanager
from typing import TYPE_CHECKING, AsyncIterator

import httpx

if TYPE_CHECKING:
    from unifi_mcp.config import UniFiConfig

logger = logging.getLogger(__name__)


class SessionManager:
    """
    Manages HTTP session lifecycle.

    Provides connection pooling and automatic cleanup.

    Usage:
        async with SessionManager(config) as session:
            client = session.client
            # Use client for requests
    """

    def __init__(self, config: UniFiConfig) -> None:
        """
        Initialize session manager.

        Args:
            config: Server configuration
        """
        self.config = config
        self._client: httpx.AsyncClient | None = None

    @property
    def client(self) -> httpx.AsyncClient:
        """Get the HTTP client (raises if not connected)."""
        if self._client is None:
            raise RuntimeError("Session not started - use async context manager")
        return self._client

    async def start(self) -> httpx.AsyncClient:
        """
        Start the HTTP session.

        Returns:
            Configured httpx.AsyncClient
        """
        if self._client is not None:
            return self._client

        logger.debug("Starting HTTP session")

        # Configure SSL verification
        verify = self.config.unifi_verify_ssl
        if not verify:
            logger.debug("SSL verification disabled")

        self._client = httpx.AsyncClient(
            base_url=self.config.unifi_host,
            verify=verify,
            timeout=httpx.Timeout(
                connect=self.config.connection_timeout,
                read=self.config.request_timeout,
                write=self.config.request_timeout,
                pool=self.config.connection_timeout,
            ),
            limits=httpx.Limits(
                max_connections=10,
                max_keepalive_connections=5,
            ),
            # Don't follow redirects automatically
            follow_redirects=False,
        )

        return self._client

    async def stop(self) -> None:
        """Close the HTTP session."""
        if self._client is not None:
            logger.debug("Closing HTTP session")
            await self._client.aclose()
            self._client = None

    async def __aenter__(self) -> "SessionManager":
        """Async context manager entry."""
        await self.start()
        return self

    async def __aexit__(self, *args: object) -> None:
        """Async context manager exit."""
        await self.stop()


@asynccontextmanager
async def create_session(config: UniFiConfig) -> AsyncIterator[httpx.AsyncClient]:
    """
    Create a managed HTTP session.

    Args:
        config: Server configuration

    Yields:
        Configured httpx.AsyncClient
    """
    manager = SessionManager(config)
    try:
        yield await manager.start()
    finally:
        await manager.stop()
```

```python
"""
Main UniFi API client.

Provides a high-level interface for all UniFi API operations.
"""

from __future__ import annotations

import logging
from typing import Any, Literal

import httpx

from unifi_mcp.config import UniFiConfig
from unifi_mcp.errors import APIError, AuthenticationError
from unifi_mcp.client.auth import UniFiAuthenticator
from unifi_mcp.client.session import SessionManager

logger = logging.getLogger(__name__)


class UniFiClient:
    """
    High-level UniFi API client.

    Handles authentication, session management, and API requests.

    Usage:
        async with UniFiClient(config) as client:
            networks = await client.get("/rest/networkconf")
            await client.post("/rest/networkconf", {"name": "New VLAN", ...})

    The client automatically handles:
    - Authentication on first request
    - Session refresh when expired
    - CSRF token management
    - Error response handling
    """

    def __init__(self, config: UniFiConfig) -> None:
        """
        Initialize the client.

        Args:
            config: Server configuration
        """
        self.config = config
        self._session_manager = SessionManager(config)
        self._auth = UniFiAuthenticator(config)
        self._connected = False

    @property
    def is_connected(self) -> bool:
        """Check if client is connected and authenticated."""
        return self._connected and self._auth.is_authenticated

    async def connect(self) -> None:
        """
        Establish connection and authenticate.

        Raises:
            AuthenticationError: If authentication fails
        """
        if self._connected:
            return

        logger.info(f"Connecting to UniFi controller at {self.config.unifi_host}")

        # Start HTTP session
        await self._session_manager.start()
        self._auth.set_http_client(self._session_manager.client)

        # Authenticate
        await self._auth.login()
        self._connected = True

        logger.info("Successfully connected to UniFi controller")

    async def disconnect(self) -> None:
        """Close connection and end session."""
        if not self._connected:
            return

        logger.info("Disconnecting from UniFi controller")

        # Logout (best effort)
        try:
            await self._auth.logout()
        except Exception as e:
            logger.debug(f"Logout error (ignored): {e}")

        # Close HTTP session
        await self._session_manager.stop()
        self._connected = False

    async def __aenter__(self) -> "UniFiClient":
        """Async context manager entry."""
        await self.connect()
        return self

    async def __aexit__(self, *args: object) -> None:
        """Async context manager exit."""
        await self.disconnect()

    async def request(
        self,
        method: Literal["GET", "POST", "PUT", "DELETE"],
        endpoint: str,
        data: dict[str, Any] | None = None,
        params: dict[str, Any] | None = None,
        retry_auth: bool = True,
    ) -> dict[str, Any]:
        """
        Make an authenticated API request.

        Args:
            method: HTTP method
            endpoint: API endpoint (e.g., "/rest/networkconf")
            data: Request body for POST/PUT
            params: Query parameters
            retry_auth: Retry with fresh auth on 401

        Returns:
            Response JSON data

        Raises:
            APIError: If request fails
            AuthenticationError: If authentication fails
        """
        if not self._connected:
            await self.connect()

        # Build full URL
        url = f"/proxy/network/api/s/{self.config.unifi_site}{endpoint}"

        # Get auth headers
        headers = self._auth.get_auth_headers()
        headers["Content-Type"] = "application/json"

        logger.debug(f"API Request: {method} {url}")

        try:
            response = await self._session_manager.client.request(
                method,
                url,
                headers=headers,
                json=data,
                params=params,
            )

            # Handle 401 - session expired
            if response.status_code == 401 and retry_auth:
                logger.warning("Session expired, re-authenticating")
                await self._auth.refresh()
                return await self.request(
                    method, endpoint, data, params, retry_auth=False
                )

            # Handle errors
            if response.status_code >= 400:
                await self._handle_error_response(response)

            # Parse response
            result = response.json()

            # UniFi API returns {"meta": {"rc": "ok/error"}, "data": [...]}
            if isinstance(result, dict):
                meta = result.get("meta", {})
                if meta.get("rc") == "error":
                    raise APIError(
                        meta.get("msg", "Unknown API error"),
                        status_code=response.status_code,
                        response_body=result,
                    )

            return result

        except httpx.ConnectError as e:
            raise APIError(
                f"Connection failed: {e}",
                status_code=None,
            ) from e
        except httpx.TimeoutException as e:
            raise APIError(
                f"Request timeout: {e}",
                status_code=None,
            ) from e

    async def _handle_error_response(self, response: httpx.Response) -> None:
        """Handle HTTP error responses."""
        try:
            body = response.json()
            message = body.get("meta", {}).get("msg", response.text)
        except Exception:
            message = response.text

        if response.status_code == 401:
            raise AuthenticationError(f"Authentication failed: {message}")
        elif response.status_code == 403:
            raise APIError(
                f"Permission denied: {message}",
                status_code=403,
            )
        elif response.status_code == 404:
            raise APIError(
                f"Resource not found: {message}",
                status_code=404,
            )
        elif response.status_code == 409:
            raise APIError(
                f"Conflict: {message}",
                status_code=409,
            )
        elif response.status_code >= 500:
            raise APIError(
                f"Server error: {message}",
                status_code=response.status_code,
            )
        else:
            raise APIError(
                f"Request failed ({response.status_code}): {message}",
                status_code=response.status_code,
            )

    # Convenience methods

    async def get(
        self,
        endpoint: str,
        params: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        """
        Make a GET request.

        Args:
            endpoint: API endpoint
            params: Query parameters

        Returns:
            Response data
        """
        return await self.request("GET", endpoint, params=params)

    async def post(
        self,
        endpoint: str,
        data: dict[str, Any],
    ) -> dict[str, Any]:
        """
        Make a POST request.

        Args:
            endpoint: API endpoint
            data: Request body

        Returns:
            Response data
        """
        return await self.request("POST", endpoint, data=data)

    async def put(
        self,
        endpoint: str,
        data: dict[str, Any],
    ) -> dict[str, Any]:
        """
        Make a PUT request.

        Args:
            endpoint: API endpoint
            data: Request body

        Returns:
            Response data
        """
        return await self.request("PUT", endpoint, data=data)

    async def delete(
        self,
        endpoint: str,
    ) -> dict[str, Any]:
        """
        Make a DELETE request.

        Args:
            endpoint: API endpoint

        Returns:
            Response data
        """
        return await self.request("DELETE", endpoint)

    # High-level helpers

    async def get_controller_info(self) -> dict[str, Any]:
        """Get UniFi controller information."""
        return await self.get("/stat/sysinfo")

    async def get_site_info(self) -> dict[str, Any]:
        """Get current site information."""
        return await self.get("/stat/sites")

    async def test_connection(self) -> bool:
        """
        Test the API connection.

        Returns:
            True if connection successful
        """
        try:
            await self.get_controller_info()
            return True
        except Exception as e:
            logger.error(f"Connection test failed: {e}")
            return False
```
