```python
"""
Configuration management for UniFi MCP Server.

Configuration sources (in order of precedence):
1. Environment variables
2. Configuration file (YAML/JSON)
3. Profile-specific settings
4. Default values

Example usage:
    # Load from environment
    config = UniFiConfig()

    # Load from file
    config = UniFiConfig.from_file("config.yaml")

    # Load specific profile
    config = UniFiConfig.from_file("config.yaml", profile="production")
"""

from __future__ import annotations

import json
import logging
import os
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Literal, Self

import yaml
from pydantic import (
    Field,
    SecretStr,
    field_validator,
    model_validator,
)
from pydantic_settings import BaseSettings, SettingsConfigDict

logger = logging.getLogger(__name__)


class ConfigurationError(Exception):
    """Configuration loading or validation error."""

    pass


@dataclass(frozen=True)
class ConnectionInfo:
    """Validated connection information."""

    host: str
    username: str
    site: str
    verify_ssl: bool
    api_base_url: str


class UniFiConfig(BaseSettings):
    """
    UniFi MCP Server Configuration.

    Configuration can be provided via:
    - Environment variables (UNIFI_HOST, UNIFI_USERNAME, etc.)
    - Configuration file (YAML or JSON)
    - Direct instantiation

    Attributes:
        unifi_host: Controller URL (e.g., https://192.168.1.1)
        unifi_username: API username
        unifi_password: API password (SecretStr for safety)
        unifi_site: Site name (default: "default")
        unifi_verify_ssl: SSL verification (default: False for self-signed)
        dry_run: Preview mode without changes (default: False)
        max_changes_per_minute: Rate limit for writes (default: 5)
        rollback_retention_minutes: Rollback data retention (default: 60)
        audit_log_path: Path for audit log (default: logs/unifi-audit.log)
        log_level: Logging verbosity (default: INFO)
    """

    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",
        validate_default=True,
    )

    # Connection settings
    unifi_host: str = Field(
        description="UniFi controller URL (e.g., https://192.168.1.1)",
    )
    unifi_username: str = Field(
        description="API username",
    )
    unifi_password: SecretStr = Field(
        description="API password",
    )
    unifi_site: str = Field(
        default="default",
        description="UniFi site name",
    )
    unifi_verify_ssl: bool = Field(
        default=False,
        description="Verify SSL certificate (disable for self-signed)",
    )

    # Timeouts
    connection_timeout: float = Field(
        default=10.0,
        ge=1.0,
        le=60.0,
        description="Connection timeout in seconds",
    )
    request_timeout: float = Field(
        default=30.0,
        ge=5.0,
        le=300.0,
        description="Request timeout in seconds",
    )

    # Safety settings
    dry_run: bool = Field(
        default=False,
        description="Preview mode - don't actually make changes",
    )
    max_changes_per_minute: int = Field(
        default=5,
        ge=1,
        le=60,
        description="Rate limit for write operations",
    )
    rollback_retention_minutes: int = Field(
        default=60,
        ge=1,
        le=1440,
        description="How long to keep rollback data (max 24 hours)",
    )
    require_confirmation_for: list[str] = Field(
        default_factory=lambda: ["medium", "high", "critical"],
        description="Permission tiers requiring confirmation",
    )

    # Logging settings
    audit_log_path: Path = Field(
        default=Path("logs/unifi-audit.log"),
        description="Path for audit log file",
    )
    log_level: Literal["DEBUG", "INFO", "WARNING", "ERROR"] = Field(
        default="INFO",
        description="Logging verbosity level",
    )
    log_to_console: bool = Field(
        default=True,
        description="Log to console in addition to file",
    )

    # Feature flags
    enable_rollback: bool = Field(
        default=True,
        description="Enable rollback functionality",
    )
    enable_audit_log: bool = Field(
        default=True,
        description="Enable audit logging",
    )

    @field_validator("unifi_host")
    @classmethod
    def validate_host(cls, v: str) -> str:
        """Ensure host is a valid URL."""
        v = v.strip()
        if not v.startswith(("http://", "https://")):
            raise ValueError("Host must start with http:// or https://")
        return v.rstrip("/")

    @field_validator("unifi_site")
    @classmethod
    def validate_site(cls, v: str) -> str:
        """Validate site name."""
        v = v.strip().lower()
        if not v:
            return "default"
        # Site names should be alphanumeric with underscores
        if not all(c.isalnum() or c == "_" for c in v):
            raise ValueError("Site name must be alphanumeric (underscores allowed)")
        return v

    @field_validator("audit_log_path")
    @classmethod
    def validate_log_path(cls, v: Path) -> Path:
        """Ensure log directory can be created."""
        if v.suffix not in (".log", ".jsonl"):
            v = v.with_suffix(".log")
        return v

    @model_validator(mode="after")
    def validate_config(self) -> Self:
        """Cross-field validation."""
        # Warn if SSL verification is disabled in non-dry-run mode
        if not self.unifi_verify_ssl and not self.dry_run:
            logger.warning(
                "SSL verification is disabled - consider enabling in production"
            )
        return self

    @property
    def api_base_url(self) -> str:
        """Get the base URL for API requests."""
        return f"{self.unifi_host}/proxy/network/api/s/{self.unifi_site}"

    @property
    def auth_url(self) -> str:
        """Get the authentication endpoint URL."""
        return f"{self.unifi_host}/api/auth/login"

    @property
    def connection_info(self) -> ConnectionInfo:
        """Get validated connection information (without secrets)."""
        return ConnectionInfo(
            host=self.unifi_host,
            username=self.unifi_username,
            site=self.unifi_site,
            verify_ssl=self.unifi_verify_ssl,
            api_base_url=self.api_base_url,
        )

    @classmethod
    def from_file(
        cls,
        path: str | Path,
        profile: str | None = None,
    ) -> "UniFiConfig":
        """
        Load configuration from a YAML or JSON file.

        Args:
            path: Path to configuration file
            profile: Optional profile name to use

        Returns:
            Loaded configuration

        Raises:
            ConfigurationError: If file cannot be loaded or is invalid
        """
        path = Path(path)

        if not path.exists():
            raise ConfigurationError(f"Configuration file not found: {path}")

        try:
            content = path.read_text()

            if path.suffix in (".yaml", ".yml"):
                data = yaml.safe_load(content) or {}
            elif path.suffix == ".json":
                data = json.loads(content)
            else:
                raise ConfigurationError(
                    f"Unsupported config format: {path.suffix} (use .yaml or .json)"
                )

            # Handle profiles
            if profile:
                if "profiles" not in data:
                    raise ConfigurationError(f"No profiles defined in {path}")
                if profile not in data["profiles"]:
                    available = list(data["profiles"].keys())
                    raise ConfigurationError(
                        f"Profile '{profile}' not found. Available: {available}"
                    )
                # Merge base config with profile
                profile_data = data["profiles"][profile]
                base_data = {k: v for k, v in data.items() if k != "profiles"}
                data = {**base_data, **profile_data}
            else:
                # Remove profiles section if present
                data.pop("profiles", None)

            # Apply environment variable overrides
            env_overrides = cls._get_env_overrides()
            data.update(env_overrides)

            return cls(**data)

        except yaml.YAMLError as e:
            raise ConfigurationError(f"Invalid YAML in {path}: {e}") from e
        except json.JSONDecodeError as e:
            raise ConfigurationError(f"Invalid JSON in {path}: {e}") from e

    @classmethod
    def _get_env_overrides(cls) -> dict[str, Any]:
        """Get configuration overrides from environment variables."""
        overrides: dict[str, Any] = {}

        env_mapping = {
            "UNIFI_HOST": "unifi_host",
            "UNIFI_USERNAME": "unifi_username",
            "UNIFI_PASSWORD": "unifi_password",
            "UNIFI_SITE": "unifi_site",
            "UNIFI_VERIFY_SSL": "unifi_verify_ssl",
            "DRY_RUN": "dry_run",
            "MAX_CHANGES_PER_MINUTE": "max_changes_per_minute",
            "ROLLBACK_RETENTION_MINUTES": "rollback_retention_minutes",
            "AUDIT_LOG_PATH": "audit_log_path",
            "LOG_LEVEL": "log_level",
        }

        for env_var, config_key in env_mapping.items():
            value = os.getenv(env_var)
            if value is not None:
                # Convert boolean strings
                if value.lower() in ("true", "false"):
                    value = value.lower() == "true"
                # Convert integer strings
                elif value.isdigit():
                    value = int(value)
                overrides[config_key] = value

        return overrides

    def to_dict(self, include_secrets: bool = False) -> dict[str, Any]:
        """
        Convert configuration to dictionary.

        Args:
            include_secrets: Include password (default False)

        Returns:
            Configuration as dictionary
        """
        data = {
            "unifi_host": self.unifi_host,
            "unifi_username": self.unifi_username,
            "unifi_site": self.unifi_site,
            "unifi_verify_ssl": self.unifi_verify_ssl,
            "connection_timeout": self.connection_timeout,
            "request_timeout": self.request_timeout,
            "dry_run": self.dry_run,
            "max_changes_per_minute": self.max_changes_per_minute,
            "rollback_retention_minutes": self.rollback_retention_minutes,
            "require_confirmation_for": self.require_confirmation_for,
            "audit_log_path": str(self.audit_log_path),
            "log_level": self.log_level,
            "log_to_console": self.log_to_console,
            "enable_rollback": self.enable_rollback,
            "enable_audit_log": self.enable_audit_log,
        }

        if include_secrets:
            data["unifi_password"] = self.unifi_password.get_secret_value()

        return data

    def __repr__(self) -> str:
        """Safe representation without secrets."""
        return (
            f"UniFiConfig(host={self.unifi_host!r}, "
            f"user={self.unifi_username!r}, "
            f"site={self.unifi_site!r})"
        )
```

```python
"""
Configuration loader with multiple source support.

Provides a unified interface for loading configuration from
environment, files, or direct values.
"""

from __future__ import annotations

import os
from pathlib import Path
from typing import Any

from unifi_mcp.config import UniFiConfig, ConfigurationError


class ConfigLoader:
    """
    Unified configuration loader.

    Usage:
        # Auto-detect configuration
        loader = ConfigLoader()
        config = loader.load()

        # Load specific file
        loader = ConfigLoader(config_file="config.yaml")
        config = loader.load(profile="production")

        # From environment only
        config = ConfigLoader.from_env()
    """

    DEFAULT_CONFIG_PATHS = [
        "config.yaml",
        "config.yml",
        "config.json",
        ".unifi-mcp.yaml",
        ".unifi-mcp.json",
    ]

    def __init__(
        self,
        config_file: str | Path | None = None,
        env_prefix: str = "UNIFI_",
    ) -> None:
        """
        Initialize the configuration loader.

        Args:
            config_file: Explicit config file path (optional)
            env_prefix: Prefix for environment variables
        """
        self.config_file = Path(config_file) if config_file else None
        self.env_prefix = env_prefix

    def load(self, profile: str | None = None) -> UniFiConfig:
        """
        Load configuration from all available sources.

        Priority:
        1. Environment variables (highest)
        2. Config file (if specified or found)
        3. Default values (lowest)

        Args:
            profile: Configuration profile to use

        Returns:
            Loaded UniFiConfig instance
        """
        # Determine config file
        config_file = self._find_config_file()

        if config_file:
            return UniFiConfig.from_file(config_file, profile=profile)
        else:
            return UniFiConfig()

    def _find_config_file(self) -> Path | None:
        """Find configuration file to use."""
        if self.config_file:
            if not self.config_file.exists():
                raise ConfigurationError(
                    f"Specified config file not found: {self.config_file}"
                )
            return self.config_file

        # Check environment for config file path
        env_config = os.getenv(f"{self.env_prefix}CONFIG_FILE")
        if env_config:
            path = Path(env_config)
            if not path.exists():
                raise ConfigurationError(
                    f"Config file from {self.env_prefix}CONFIG_FILE not found: {path}"
                )
            return path

        # Search default locations
        for default_path in self.DEFAULT_CONFIG_PATHS:
            path = Path(default_path)
            if path.exists():
                return path

        return None

    @classmethod
    def from_env(cls) -> UniFiConfig:
        """Load configuration from environment variables only."""
        return UniFiConfig()

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> UniFiConfig:
        """Load configuration from a dictionary."""
        return UniFiConfig(**data)


def get_config(
    config_file: str | Path | None = None,
    profile: str | None = None,
) -> UniFiConfig:
    """
    Convenience function to load configuration.

    Args:
        config_file: Optional path to config file
        profile: Optional profile name

    Returns:
        Loaded configuration
    """
    loader = ConfigLoader(config_file=config_file)
    return loader.load(profile=profile)
```

```python
"""
Secret management helpers.

Provides utilities for safely handling credentials and secrets.
Currently supports:
- Environment variables
- .env files
- Placeholder expansion

Future extension points for:
- HashiCorp Vault
- AWS Secrets Manager
- 1Password CLI
"""

from __future__ import annotations

import os
import re
from pathlib import Path
from typing import Any


class SecretResolutionError(Exception):
    """Failed to resolve a secret."""

    pass


def resolve_env_placeholder(value: str) -> str:
    """
    Resolve environment variable placeholders in a string.

    Supports ${VAR_NAME} and $VAR_NAME syntax.

    Args:
        value: String potentially containing placeholders

    Returns:
        String with placeholders resolved

    Raises:
        SecretResolutionError: If required env var is not set
    """
    # Match ${VAR_NAME} or ${VAR_NAME:-default}
    pattern = r"\$\{([A-Z_][A-Z0-9_]*)(?::-([^}]*))?\}"

    def replace(match: re.Match[str]) -> str:
        var_name = match.group(1)
        default = match.group(2)

        env_value = os.getenv(var_name)
        if env_value is not None:
            return env_value
        if default is not None:
            return default
        raise SecretResolutionError(
            f"Environment variable {var_name} is not set"
        )

    return re.sub(pattern, replace, value)


def resolve_secrets(data: dict[str, Any]) -> dict[str, Any]:
    """
    Recursively resolve secret placeholders in configuration data.

    Args:
        data: Configuration dictionary

    Returns:
        Dictionary with secrets resolved
    """
    result: dict[str, Any] = {}

    for key, value in data.items():
        if isinstance(value, str):
            result[key] = resolve_env_placeholder(value)
        elif isinstance(value, dict):
            result[key] = resolve_secrets(value)
        elif isinstance(value, list):
            result[key] = [
                resolve_secrets(item) if isinstance(item, dict)
                else resolve_env_placeholder(item) if isinstance(item, str)
                else item
                for item in value
            ]
        else:
            result[key] = value

    return result


def mask_secret(value: str, visible_chars: int = 4) -> str:
    """
    Mask a secret value for safe logging.

    Args:
        value: Secret value to mask
        visible_chars: Number of characters to show at end

    Returns:
        Masked string (e.g., "****cret")
    """
    if len(value) <= visible_chars:
        return "*" * len(value)
    return "*" * (len(value) - visible_chars) + value[-visible_chars:]


def load_env_file(path: str | Path = ".env") -> dict[str, str]:
    """
    Load environment variables from a .env file.

    Args:
        path: Path to .env file

    Returns:
        Dictionary of loaded variables
    """
    path = Path(path)
    if not path.exists():
        return {}

    env_vars: dict[str, str] = {}
    content = path.read_text()

    for line in content.splitlines():
        line = line.strip()
        # Skip comments and empty lines
        if not line or line.startswith("#"):
            continue

        # Parse KEY=VALUE
        if "=" in line:
            key, _, value = line.partition("=")
            key = key.strip()
            value = value.strip()

            # Remove quotes
            if (value.startswith('"') and value.endswith('"')) or \
               (value.startswith("'") and value.endswith("'")):
                value = value[1:-1]

            env_vars[key] = value

    return env_vars


def inject_env_file(path: str | Path = ".env") -> None:
    """
    Load .env file and inject into os.environ.

    Args:
        path: Path to .env file
    """
    env_vars = load_env_file(path)
    for key, value in env_vars.items():
        if key not in os.environ:  # Don't override existing
            os.environ[key] = value
```
