## 1. Objective

Implement a command-line interface that orchestrates the crawler, bringing together all components (fetcher, adapters, converters, state) into a usable tool.

## 2. Prerequisites

- Tasks F1.1-F1.7 completed
- All components working and tested
- Note: F1.4 (Guides Adapter) is deferred; `--source guides` disabled

## 3. Scope

### In Scope
- `crawl` command (full and incremental modes)
- `status` command (show crawl state)
- `verify` command (check for updates without downloading)
- Progress display with rich library
- Configurable output directory
- Error reporting and summaries
- Automatic index manifest generation after crawl
- Semantic file organization by topic

### Out of Scope
- Web UI
- Daemon mode / scheduler
- Parallel crawling (keep sequential for simplicity)
- Embedding generation (not needed for agentic use)

**Key Design Decisions:**
- Documents organized by topic (not source URL structure)
- Semantic file names (not article IDs)
- Single index.yaml for discoverability
- Flat topic directories (no deep nesting)

```python
"""
Command-line interface for UniFi documentation crawler.
"""

import asyncio
from pathlib import Path
from typing import Optional

import click
from rich.console import Console
from rich.progress import Progress, SpinnerColumn, TextColumn, BarColumn, TaskProgressColumn
from rich.table import Table

from doc_crawler.config import CrawlerConfig
from doc_crawler.crawler import Crawler
from doc_crawler.state import StateManager

console = Console()

@click.group()
@click.option("--config", "-c", type=click.Path(exists=True), help="Path to config file")
@click.option("--verbose", "-v", is_flag=True, help="Verbose output")
@click.pass_context
def cli(ctx: click.Context, config: Optional[str], verbose: bool):
    """UniFi Documentation Crawler - Fetch and organize UniFi docs for offline use."""
    ctx.ensure_object(dict)

    # Load config
    config_path = Path(config) if config else Path("config.yaml")
    ctx.obj["config"] = CrawlerConfig.from_yaml(config_path)
    ctx.obj["verbose"] = verbose

@cli.command()
@click.option("--mode", "-m", type=click.Choice(["full", "incremental"]), default="incremental")
@click.option("--source", "-s", type=click.Choice(["help-center"]), default="help-center",
              help="Source to crawl (guides deferred to later phase)")
@click.option("--max-items", type=int, default=None, help="Maximum items to process (for testing)")
@click.option("--dry-run", is_flag=True, help="Show what would be crawled without downloading")
@click.pass_context
def crawl(ctx: click.Context, mode: str, source: str, max_items: int | None, dry_run: bool):
    """
    Crawl documentation sources.

    --mode full: Re-crawl everything
    --mode incremental: Only new/changed content (default)
    """
    config = ctx.obj["config"]
    verbose = ctx.obj["verbose"]

    console.print(f"[bold blue]UniFi Documentation Crawler[/bold blue]")
    console.print(f"Mode: {mode}, Source: {source}")

    if dry_run:
        console.print("[yellow]DRY RUN - No files will be downloaded[/yellow]")

    asyncio.run(_run_crawl(config, mode, source, max_items, dry_run, verbose))

async def _run_crawl(
    config: CrawlerConfig,
    mode: str,
    source: str,
    max_items: int | None,
    dry_run: bool,
    verbose: bool,
):
    """Run the crawl operation with index generation."""
    from doc_crawler.index import IndexBuilder
    from doc_crawler.paths import generate_semantic_path, classify_topic, extract_keywords

    state = StateManager(config.output_dir / config.state_file)
    state.load()

    # Initialize index builder
    index_builder = IndexBuilder(config.output_dir)

    # Determine sources to crawl (guides deferred)
    sources = []
    if source == "help-center" and config.sources.help_center.enabled:
        sources.append("help_center")
    # Note: guides source disabled pending F1.4 implementation

    if not sources:
        console.print("[yellow]No sources enabled in configuration[/yellow]")
        return

    total_discovered = 0
    total_crawled = 0
    total_skipped = 0
    total_failed = 0

    state.start_crawl(full=(mode == "full"))

    for src_name in sources:
        console.print(f"\n[bold]Crawling: {src_name}[/bold]")

        with Progress(
            SpinnerColumn(),
            TextColumn("[progress.description]{task.description}"),
            BarColumn(),
            TaskProgressColumn(),
            console=console,
        ) as progress:
            # Discovery phase
            discover_task = progress.add_task("Discovering...", total=None)
            items = []

            async for item in crawler.discover(src_name):
                items.append(item)
                progress.update(discover_task, description=f"Discovered {len(items)} items")

            progress.update(discover_task, completed=True)
            total_discovered += len(items)

            # Filter for incremental mode
            if mode == "incremental":
                items_to_crawl = [
                    item for item in items
                    if state.needs_crawl(item.url)
                ]
                skipped = len(items) - len(items_to_crawl)
                total_skipped += skipped
                if verbose and skipped > 0:
                    console.print(f"  Skipping {skipped} unchanged items")
            else:
                items_to_crawl = items

            # Apply max_items limit
            if max_items is not None:
                items_to_crawl = items_to_crawl[:max_items]

            if not items_to_crawl:
                console.print("  [green]No new items to crawl[/green]")
                continue

            if dry_run:
                console.print(f"  Would crawl {len(items_to_crawl)} items")
                for item in items_to_crawl[:5]:
                    console.print(f"    - {item.metadata.get('title', item.url)}")
                if len(items_to_crawl) > 5:
                    console.print(f"    ... and {len(items_to_crawl) - 5} more")
                continue

            # Crawl phase
            crawl_task = progress.add_task("Crawling...", total=len(items_to_crawl))

            for item in items_to_crawl:
                try:
                    result = await crawler.fetch_and_save(item)

                    if result.success:
                        state.record_success(
                            url=item.url,
                            checksum=result.checksum,
                            source=src_name,
                        )

                        # Add to index manifest
                        title = item.metadata.get("title", "Untitled")
                        result_meta = result.metadata or {}
                        topic = result_meta.get("topic") or classify_topic(title, "")
                        index_builder.add_document(
                            path=result.path.relative_to(config.output_dir).as_posix() if result.path else "",
                            title=title,
                            overview=result_meta.get("overview", ""),
                            keywords=extract_keywords(title, ""),
                            topic=topic,
                            original_id=item.metadata.get("article_id", ""),
                            url=item.url,
                            updated_at=item.metadata.get("updated_at"),
                            checksum=result.checksum or "",
                        )

                        total_crawled += 1
                    else:
                        state.record_failure(item.url, src_name, result.error or "Unknown error")
                        total_failed += 1
                        if verbose:
                            console.print(f"  [red]Failed: {item.url}[/red]")

                except Exception as e:
                    state.record_failure(item.url, src_name, str(e))
                    total_failed += 1
                    if verbose:
                        console.print(f"  [red]Error: {e}[/red]")

                progress.update(crawl_task, advance=1)

        state.finish_source(src_name, total_crawled)

    # Save final state
    state.save(force=True)

    # Generate index manifest
    if total_crawled > 0:
        console.print("\n[bold]Generating index manifest...[/bold]")
        index_builder.finalize(source="help.ui.com")
        index_path = index_builder.save()
        console.print(f"  Index saved to: {index_path}")

    # Summary
    console.print("\n[bold]Summary:[/bold]")
    table = Table()
    table.add_column("Metric", style="cyan")
    table.add_column("Count", justify="right")
    table.add_row("Discovered", str(total_discovered))
    table.add_row("Crawled", str(total_crawled))
    table.add_row("Skipped (unchanged)", str(total_skipped))
    table.add_row("Failed", str(total_failed))
    if index_builder.manifest.document_count > 0:
        table.add_row("Topics", str(len(index_builder.manifest.topics)))
    console.print(table)

@cli.command()
@click.pass_context
def status(ctx: click.Context):
    """Show crawl status and statistics."""
    config = ctx.obj["config"]
    state = StateManager(config.output_dir / config.state_file)

    if not state.load():
        console.print("[yellow]No crawl state found. Run 'crawl' first.[/yellow]")
        return

    stats = state.get_stats()

    console.print("[bold]Crawl Status[/bold]\n")

    table = Table()
    table.add_column("Metric", style="cyan")
    table.add_column("Value")

    table.add_row("Total items", str(stats["total_items"]))
    table.add_row("Successful", str(stats["successful"]))
    table.add_row("Failed", str(stats["failed"]))
    table.add_row("Last full crawl", stats["last_full_crawl"] or "Never")
    table.add_row("Last incremental crawl", stats["last_incremental_crawl"] or "Never")
    table.add_row("Sources", ", ".join(stats["sources"]) or "None")

    console.print(table)

@cli.command()
@click.option("--source", "-s", type=click.Choice(["help-center"]), default="help-center",
              help="Source to verify (guides deferred to later phase)")
@click.pass_context
def verify(ctx: click.Context, source: str):
    """Check for updates without downloading."""
    config = ctx.obj["config"]
    state = StateManager(config.output_dir / config.state_file)
    state.load()

    console.print("[bold]Checking for updates...[/bold]\n")

    asyncio.run(_run_verify(config, state, source))

async def _run_verify(config: CrawlerConfig, state: StateManager, source: str):
    """Run verification check."""
    # Determine sources to verify (guides deferred)
    sources = []
    if source == "help-center" and config.sources.help_center.enabled:
        sources.append("help_center")
    # Note: guides source disabled pending F1.4 implementation

    async with Crawler(config) as crawler:
        for src_name in sources:
            console.print(f"[bold]{src_name}:[/bold]")

            new_count = 0
            changed_count = 0

            async for item in crawler.discover(src_name):
                item_state = state.get_item_state(item.url)

                if not item_state:
                    new_count += 1
                elif state.needs_crawl(item.url):
                    changed_count += 1

            if new_count > 0:
                console.print(f"  [green]{new_count} new items[/green]")
            if changed_count > 0:
                console.print(f"  [yellow]{changed_count} items need re-crawl[/yellow]")
            if new_count == 0 and changed_count == 0:
                console.print("  [dim]No updates found[/dim]")
```

### 4.3 Main Crawler Orchestrator

**src/crawler.py:**

```python
"""
Main crawler orchestrator.
"""

from dataclasses import dataclass
from pathlib import Path
from typing import AsyncIterator, Optional

import hashlib
from doc_crawler.config import CrawlerConfig
from doc_crawler.fetcher import Fetcher
from doc_crawler.adapters.base import CrawlItem, CrawlResult
from doc_crawler.adapters.help_center import HelpCenterAdapter
# from doc_crawler.adapters.guides import GuidesAdapter  # Deferred (F1.4)
from doc_crawler.converters.html import HTMLConverter
from doc_crawler.converters.pdf import PDFConverter

@dataclass
class SaveResult:
    """Result of saving a crawled item."""
    success: bool
    path: Optional[Path] = None
    checksum: Optional[str] = None
    error: Optional[str] = None
    metadata: Optional[dict] = None  # For passing to index builder

class Crawler:
    """Main crawler that orchestrates all components."""

    def __init__(self, config: CrawlerConfig):
        self.config = config
        self._fetcher: Optional[Fetcher] = None
        self._html_converter = HTMLConverter()
        self._pdf_converter = PDFConverter()

    async def __aenter__(self) -> "Crawler":
        self._fetcher = Fetcher(self.config)
        await self._fetcher.__aenter__()
        return self

    async def __aexit__(self, *args):
        if self._fetcher:
            await self._fetcher.__aexit__(*args)

    def _get_adapter(self, source: str):
        """Get adapter for source name."""
        if not self._fetcher:
            raise RuntimeError("Crawler must be used as context manager")

        if source == "help_center":
            return HelpCenterAdapter(self.config.sources.help_center, self._fetcher)
        elif source == "guides":
            # GuidesAdapter deferred (F1.4)
            raise NotImplementedError("Guides adapter deferred to later phase (F1.4)")
        else:
            raise ValueError(f"Unknown source: {source}")

    async def discover(self, source: str) -> AsyncIterator[CrawlItem]:
        """Discover items from a source."""
        async with Fetcher(self.config) as fetcher:
            adapter = self._get_adapter_with_fetcher(source, fetcher)
            async for item in adapter.discover():
                yield item

    def _get_adapter_with_fetcher(self, source: str, fetcher: Fetcher):
        """Get adapter with specific fetcher instance."""
        if source == "help_center":
            return HelpCenterAdapter(self.config.sources.help_center, fetcher)
        elif source == "guides":
            # GuidesAdapter deferred (F1.4)
            raise NotImplementedError("Guides adapter deferred to later phase (F1.4)")
        else:
            raise ValueError(f"Unknown source: {source}")

    async def fetch_and_save(self, item: CrawlItem) -> SaveResult:
        """Fetch an item and save to disk."""
        async with Fetcher(self.config) as fetcher:
            adapter = self._get_adapter_with_fetcher(item.metadata.get("source", "help_center"), fetcher)
            result = await adapter.fetch(item)

            if not result.success:
                return SaveResult(success=False, error=result.error)

            # Convert and save based on type
            if item.item_type == "article":
                return await self._save_article(item, result)
            elif item.item_type == "pdf":
                return await self._save_pdf(item, result)
            else:
                return SaveResult(success=False, error=f"Unknown item type: {item.item_type}")

    async def _save_article(self, item: CrawlItem, result: CrawlResult) -> SaveResult:
        """Save HTML article as Markdown with semantic path."""
        from doc_crawler.paths import generate_semantic_path, classify_topic

        # Decode bytes to string if needed
        content = result.content
        if isinstance(content, bytes):
            content = content.decode("utf-8", errors="replace")

        # Convert to markdown (includes structured document generation)
        conversion = self._html_converter.convert(content, result.metadata)

        if not conversion.success:
            return SaveResult(success=False, error=conversion.error)

        # Generate semantic output path
        article_id = item.metadata.get("article_id", "unknown")
        title = result.metadata.get("title", "untitled")
        semantic_path = generate_semantic_path(
            article_id=article_id,
            title=title,
            content=content,
            base_dir="help-center",
        )
        output_path = self.config.output_dir / semantic_path

        # Create directory
        output_path.parent.mkdir(parents=True, exist_ok=True)

        # Write file
        output_path.write_text(conversion.markdown)

        # Store metadata for index
        result.metadata["overview"] = conversion.frontmatter.get("overview", "")
        result.metadata["topic"] = classify_topic(title, content)

        # Compute checksum from markdown content
        checksum = hashlib.md5(conversion.markdown.encode()).hexdigest()

        return SaveResult(
            success=True,
            path=output_path,
            checksum=checksum,
            metadata=result.metadata,
        )

    async def _save_pdf(self, item: CrawlItem, result: CrawlResult) -> SaveResult:
        """Save PDF and optionally extract text."""
        filename = item.metadata.get("filename", "document.pdf")

        # Save original PDF
        pdf_dir = self.config.output_dir / "guides" / "pdf"
        pdf_dir.mkdir(parents=True, exist_ok=True)
        pdf_path = pdf_dir / filename
        pdf_path.write_bytes(result.content)

        # Convert to markdown
        conversion = self._pdf_converter.convert(result.content, item.metadata)

        if conversion.success:
            md_dir = self.config.output_dir / "guides" / "markdown"
            md_dir.mkdir(parents=True, exist_ok=True)
            md_path = md_dir / filename.replace(".pdf", ".md")
            md_path.write_text(conversion.markdown)

        # Compute checksum from PDF bytes
        checksum = hashlib.md5(result.content).hexdigest()

        return SaveResult(
            success=True,
            path=pdf_path,
            checksum=checksum,
        )

    # High-level crawl methods for programmatic use

    async def crawl_help_center(self, max_articles: int | None = None) -> "CrawlStats":
        """
        Crawl Help Center articles.

        Args:
            max_articles: Maximum number of articles to process (None for all)

        Returns:
            CrawlStats with results
        """
        from doc_crawler.state import StateManager

        state = StateManager(self.config.output_dir / self.config.state_file)
        state.load()

        stats = CrawlStats()
        count = 0

        async for item in self.discover("help_center"):
            if max_articles is not None and count >= max_articles:
                break

            if not state.needs_crawl(item.url):
                stats.skipped += 1
                continue

            try:
                result = await self.fetch_and_save(item)
                if result.success:
                    state.record_success(item.url, result.checksum or "", "help_center")
                    stats.articles_processed += 1
                else:
                    state.record_failure(item.url, "help_center", result.error or "Unknown error")
                    stats.errors += 1
            except Exception as e:
                state.record_failure(item.url, "help_center", str(e))
                stats.errors += 1

            count += 1

        state.save(force=True)
        return stats

    async def crawl_all(
        self,
        max_help_center_articles: int | None = None,
        max_guide_files: int | None = None,
    ) -> "CombinedCrawlStats":
        """
        Crawl all enabled sources.

        Note: Guides source is deferred (F1.4).
        """
        help_center_stats = await self.crawl_help_center(max_articles=max_help_center_articles)

        # Guides deferred - return empty stats
        guides_stats = CrawlStats()

        return CombinedCrawlStats(
            help_center=help_center_stats,
            guides=guides_stats,
        )


@dataclass
class CrawlStats:
    """Statistics from a crawl operation."""
    articles_processed: int = 0
    files_processed: int = 0
    skipped: int = 0
    errors: int = 0


@dataclass
class CombinedCrawlStats:
    """Statistics from crawling multiple sources."""
    help_center: CrawlStats
    guides: CrawlStats

    @property
    def total_items(self) -> int:
        return (
            self.help_center.articles_processed +
            self.guides.files_processed
        )
```
