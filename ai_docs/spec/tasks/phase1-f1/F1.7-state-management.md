## 1. Objective

Implement state management for incremental crawling - track crawled URLs, checksums, and timestamps to enable efficient updates without re-crawling unchanged content.

## 2. Prerequisites

- Task F1.1 (Project Setup) completed

## 3. Scope

### In Scope
- Persist crawl state to JSON file
- Track URL → checksum mapping
- Track last crawl timestamp per URL
- Support incremental crawls (new/changed only)
- Checkpoint during long crawls
- Resume interrupted crawls
- Generate index manifest (index.yaml) for document discoverability
- Track topic statistics in index

### Out of Scope
- Database storage (JSON is sufficient)
- Distributed state (single machine)
- Content deduplication across URLs
- Embedding generation (not needed for agentic frameworks)

### 4.2 State Manager Implementation

**src/doc_crawler/state.py:**

```python
"""
State management for incremental crawling.
"""

import json
from dataclasses import dataclass, field, asdict
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional
from enum import Enum

class CrawlStatus(str, Enum):
    """Status of a crawled item."""
    SUCCESS = "success"
    FAILED = "failed"
    SKIPPED = "skipped"

@dataclass
class ItemState:
    """State of a single crawled item."""
    checksum: str
    last_crawled: str
    source: str
    status: CrawlStatus = CrawlStatus.SUCCESS
    last_modified: Optional[str] = None
    error: Optional[str] = None

@dataclass
class SourceState:
    """State of a documentation source."""
    last_crawl: str
    items_count: int = 0

@dataclass
class CrawlState:
    """Complete crawl state."""
    version: str = "1.0"
    last_full_crawl: Optional[str] = None
    last_incremental_crawl: Optional[str] = None
    sources: dict[str, SourceState] = field(default_factory=dict)
    items: dict[str, ItemState] = field(default_factory=dict)

class StateManager:
    """
    Manage crawl state for incremental updates.

    Tracks:
    - Which URLs have been crawled
    - Content checksums for change detection
    - Timestamps for freshness
    - Crawl status (success/failed)
    """

    def __init__(self, state_file: Path, save_interval: int = 50):
        """
        Initialize state manager.

        Args:
            state_file: Path to state JSON file
            save_interval: Save state every N items
        """
        self.state_file = Path(state_file)
        self.save_interval = save_interval
        self._state: CrawlState = CrawlState()
        self._dirty = False
        self._items_since_save = 0

    def load(self) -> bool:
        """
        Load state from file.

        Returns:
            True if state loaded, False if new state created
        """
        if not self.state_file.exists():
            self._state = CrawlState()
            return False

        try:
            with open(self.state_file) as f:
                data = json.load(f)

            # Parse sources
            sources = {}
            for name, src_data in data.get("sources", {}).items():
                sources[name] = SourceState(**src_data)

            # Parse items
            items = {}
            for url, item_data in data.get("items", {}).items():
                item_data["status"] = CrawlStatus(item_data.get("status", "success"))
                items[url] = ItemState(**item_data)

            self._state = CrawlState(
                version=data.get("version", "1.0"),
                last_full_crawl=data.get("last_full_crawl"),
                last_incremental_crawl=data.get("last_incremental_crawl"),
                sources=sources,
                items=items,
            )
            return True

        except (json.JSONDecodeError, KeyError, TypeError) as e:
            # Corrupted state - start fresh
            self._state = CrawlState()
            return False

    def save(self, force: bool = False) -> None:
        """
        Save state to file.

        Args:
            force: Save even if not dirty
        """
        if not self._dirty and not force:
            return

        # Convert to serializable dict
        data = {
            "version": self._state.version,
            "last_full_crawl": self._state.last_full_crawl,
            "last_incremental_crawl": self._state.last_incremental_crawl,
            "sources": {
                name: asdict(src) for name, src in self._state.sources.items()
            },
            "items": {
                url: {**asdict(item), "status": item.status.value}
                for url, item in self._state.items.items()
            },
        }

        # Write atomically
        temp_file = self.state_file.with_suffix(".tmp")
        with open(temp_file, "w") as f:
            json.dump(data, f, indent=2)
        temp_file.rename(self.state_file)

        self._dirty = False
        self._items_since_save = 0

    def _maybe_checkpoint(self) -> None:
        """Save checkpoint if enough items processed."""
        self._items_since_save += 1
        if self._items_since_save >= self.save_interval:
            self.save()

    def get_item_state(self, url: str) -> Optional[ItemState]:
        """Get state for a specific URL."""
        return self._state.items.get(url)

    def has_changed(self, url: str, new_checksum: str) -> bool:
        """
        Check if content has changed since last crawl.

        Args:
            url: URL of the item
            new_checksum: Checksum of new content

        Returns:
            True if content is new or changed
        """
        existing = self._state.items.get(url)
        if not existing:
            return True  # New URL
        return existing.checksum != new_checksum

    def needs_crawl(
        self,
        url: str,
        max_age_hours: Optional[int] = None,
    ) -> bool:
        """
        Check if URL needs to be crawled.

        Args:
            url: URL to check
            max_age_hours: Consider stale after this many hours

        Returns:
            True if URL should be crawled
        """
        existing = self._state.items.get(url)

        if not existing:
            return True  # Never crawled

        if existing.status == CrawlStatus.FAILED:
            return True  # Retry failed items

        if max_age_hours is None:
            return False  # Already crawled, no age limit

        # Check age
        try:
            last_crawled = datetime.fromisoformat(existing.last_crawled.replace("Z", "+00:00"))
            age = datetime.now(timezone.utc) - last_crawled
            return age.total_seconds() > (max_age_hours * 3600)
        except (ValueError, AttributeError):
            return True

    def record_success(
        self,
        url: str,
        checksum: str,
        source: str,
        last_modified: Optional[str] = None,
    ) -> None:
        """Record successful crawl of an item."""
        self._state.items[url] = ItemState(
            checksum=checksum,
            last_crawled=datetime.now(timezone.utc).isoformat(),
            source=source,
            status=CrawlStatus.SUCCESS,
            last_modified=last_modified,
        )
        self._dirty = True
        self._maybe_checkpoint()

    def record_failure(
        self,
        url: str,
        source: str,
        error: str,
    ) -> None:
        """Record failed crawl of an item."""
        existing = self._state.items.get(url)
        self._state.items[url] = ItemState(
            checksum=existing.checksum if existing else "",
            last_crawled=datetime.now(timezone.utc).isoformat(),
            source=source,
            status=CrawlStatus.FAILED,
            error=error,
        )
        self._dirty = True
        self._maybe_checkpoint()

    def record_skipped(self, url: str) -> None:
        """Record that item was skipped (unchanged)."""
        existing = self._state.items.get(url)
        if existing:
            existing.last_crawled = datetime.now(timezone.utc).isoformat()
            existing.status = CrawlStatus.SKIPPED
            self._dirty = True

    def start_crawl(self, full: bool = False) -> None:
        """Mark start of a crawl session."""
        now = datetime.now(timezone.utc).isoformat()
        if full:
            self._state.last_full_crawl = now
        else:
            self._state.last_incremental_crawl = now
        self._dirty = True

    def finish_source(self, source: str, items_count: int) -> None:
        """Mark completion of crawling a source."""
        self._state.sources[source] = SourceState(
            last_crawl=datetime.now(timezone.utc).isoformat(),
            items_count=items_count,
        )
        self._dirty = True

    def get_stats(self) -> dict:
        """Get crawl statistics."""
        items = self._state.items
        return {
            "total_items": len(items),
            "successful": sum(1 for i in items.values() if i.status == CrawlStatus.SUCCESS),
            "failed": sum(1 for i in items.values() if i.status == CrawlStatus.FAILED),
            "sources": list(self._state.sources.keys()),
            "last_full_crawl": self._state.last_full_crawl,
            "last_incremental_crawl": self._state.last_incremental_crawl,
        }

    def get_changed_urls(self, checksums: dict[str, str]) -> list[str]:
        """
        Given new checksums, return URLs that have changed.

        Args:
            checksums: Dict of URL -> new checksum

        Returns:
            List of URLs with changed or new content
        """
        changed = []
        for url, new_checksum in checksums.items():
            if self.has_changed(url, new_checksum):
                changed.append(url)
        return changed

    def remove_item(self, url: str) -> bool:
        """Remove an item from state (e.g., if no longer exists)."""
        if url in self._state.items:
            del self._state.items[url]
            self._dirty = True
            return True
        return False

    def clear(self) -> None:
        """Clear all state."""
        self._state = CrawlState()
        self._dirty = True
```

**src/doc_crawler/index.py:**

```python
"""
Index manifest generation and management for documentation corpus.

The index manifest enables agentic frameworks like Claude Code to quickly
discover available documentation without traversing the file system.
"""

from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional
import yaml

from .paths import TOPIC_DESCRIPTIONS

@dataclass
class DocumentEntry:
    """Entry for a single document in the index."""
    path: str
    title: str
    overview: str
    keywords: list[str]
    topic: str
    original_id: str
    url: str
    updated_at: Optional[str]
    checksum: str

@dataclass
class TopicSummary:
    """Summary of a topic category."""
    name: str
    description: str
    document_count: int = 0

@dataclass
class IndexManifest:
    """Complete index manifest."""
    version: str = "1.0"
    generated_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    source: str = ""
    document_count: int = 0
    topics: dict[str, TopicSummary] = field(default_factory=dict)
    documents: list[DocumentEntry] = field(default_factory=list)

class IndexBuilder:
    """
    Builds and maintains the index manifest.

    Usage:
        builder = IndexBuilder(output_dir)
        builder.add_document(...)
        builder.add_document(...)
        builder.finalize("help.ui.com")
        builder.save()
    """

    def __init__(self, output_dir: Path):
        self.output_dir = Path(output_dir)
        self.manifest = IndexManifest()
        self._topic_counts: dict[str, int] = {}

    def add_document(
        self,
        path: str,
        title: str,
        overview: str,
        keywords: list[str],
        topic: str,
        original_id: str,
        url: str,
        updated_at: Optional[str],
        checksum: str,
    ) -> None:
        """
        Add a document to the index.

        Args:
            path: Relative path to the document
            title: Document title
            overview: Brief overview/summary
            keywords: List of keywords
            topic: Topic category
            original_id: Original article/document ID
            url: Source URL
            updated_at: Last updated date
            checksum: Content checksum
        """
        entry = DocumentEntry(
            path=path,
            title=title,
            overview=overview[:300] + "..." if len(overview) > 300 else overview,
            keywords=keywords[:10],  # Limit keywords
            topic=topic,
            original_id=original_id,
            url=url,
            updated_at=updated_at,
            checksum=checksum,
        )
        self.manifest.documents.append(entry)

        # Track topic counts
        self._topic_counts[topic] = self._topic_counts.get(topic, 0) + 1

    def finalize(self, source: str) -> None:
        """
        Finalize the manifest with counts and topic summaries.

        Args:
            source: Source identifier (e.g., "help.ui.com")
        """
        self.manifest.source = source
        self.manifest.document_count = len(self.manifest.documents)
        self.manifest.generated_at = datetime.now(timezone.utc).isoformat()

        # Build topic summaries
        for topic, count in self._topic_counts.items():
            self.manifest.topics[topic] = TopicSummary(
                name=topic,
                description=TOPIC_DESCRIPTIONS.get(topic, f"Documentation about {topic}"),
                document_count=count,
            )

    def save(self, filename: str = "index.yaml") -> Path:
        """
        Save manifest to YAML file.

        Args:
            filename: Output filename

        Returns:
            Path to saved file
        """
        output_path = self.output_dir / filename

        # Convert to dict for YAML serialization
        data = {
            "version": self.manifest.version,
            "generated_at": self.manifest.generated_at,
            "source": self.manifest.source,
            "document_count": self.manifest.document_count,
            "topics": {
                name: {
                    "description": t.description,
                    "document_count": t.document_count,
                }
                for name, t in self.manifest.topics.items()
            },
            "documents": [
                {
                    "path": d.path,
                    "title": d.title,
                    "overview": d.overview,
                    "keywords": d.keywords,
                    "topic": d.topic,
                    "original_id": d.original_id,
                    "url": d.url,
                    "updated_at": d.updated_at,
                    "checksum": d.checksum,
                }
                for d in self.manifest.documents
            ],
        }

        # Ensure directory exists
        output_path.parent.mkdir(parents=True, exist_ok=True)

        with open(output_path, "w", encoding="utf-8") as f:
            yaml.dump(data, f, default_flow_style=False, allow_unicode=True, sort_keys=False)

        return output_path

    @classmethod
    def load(cls, filepath: Path) -> IndexManifest:
        """
        Load existing manifest from file.

        Args:
            filepath: Path to index.yaml

        Returns:
            Loaded IndexManifest
        """
        with open(filepath, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)

        manifest = IndexManifest(
            version=data.get("version", "1.0"),
            generated_at=data.get("generated_at", ""),
            source=data.get("source", ""),
            document_count=data.get("document_count", 0),
        )

        for name, topic_data in data.get("topics", {}).items():
            manifest.topics[name] = TopicSummary(
                name=name,
                description=topic_data.get("description", ""),
                document_count=topic_data.get("document_count", 0),
            )

        for doc_data in data.get("documents", []):
            manifest.documents.append(DocumentEntry(
                path=doc_data["path"],
                title=doc_data["title"],
                overview=doc_data.get("overview", ""),
                keywords=doc_data.get("keywords", []),
                topic=doc_data.get("topic", "general"),
                original_id=doc_data.get("original_id", ""),
                url=doc_data.get("url", ""),
                updated_at=doc_data.get("updated_at"),
                checksum=doc_data.get("checksum", ""),
            ))

        return manifest

    def get_documents_by_topic(self, topic: str) -> list[DocumentEntry]:
        """Get all documents for a specific topic."""
        return [d for d in self.manifest.documents if d.topic == topic]

    def search_by_keyword(self, keyword: str) -> list[DocumentEntry]:
        """Search documents by keyword."""
        keyword = keyword.lower()
        return [
            d for d in self.manifest.documents
            if keyword in d.keywords or keyword in d.title.lower()
        ]
```
