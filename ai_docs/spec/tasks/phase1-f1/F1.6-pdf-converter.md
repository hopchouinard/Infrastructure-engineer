## 1. Objective

Implement a PDF text extractor that extracts readable text from downloaded PDF documentation files, producing Markdown output with metadata.

## 2. Prerequisites

- Task F1.1 (Project Setup) completed
- `pypdf` library available (from pyproject.toml)

## 3. Scope

### In Scope
- Extract text from PDF files
- Generate basic structure (headings, paragraphs)
- Generate YAML frontmatter with metadata
- Extract PDF metadata (title, author, creation date)
- Handle multi-page documents
- Compute checksum for change detection

### Out of Scope
- OCR for scanned documents
- Image extraction
- Complex layout preservation
- Table extraction (best-effort only)
- Form field extraction

### 4.2 Converter Implementation

**src/doc_crawler/converters/pdf.py:**

```python
"""
PDF text extractor for documentation files.
"""

import hashlib
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional, Union
import re

from pypdf import PdfReader
import yaml

@dataclass
class PDFMetadata:
    """Metadata extracted from PDF."""
    title: Optional[str] = None
    author: Optional[str] = None
    subject: Optional[str] = None
    creator: Optional[str] = None
    producer: Optional[str] = None
    creation_date: Optional[str] = None
    modification_date: Optional[str] = None
    pages: int = 0

@dataclass
class PDFConversionResult:
    """Result of PDF text extraction."""
    text: str
    markdown: str
    metadata: PDFMetadata
    frontmatter: dict
    success: bool
    error: Optional[str] = None
    warnings: list[str] = field(default_factory=list)

@dataclass
class PDFConverterConfig:
    """Configuration for PDF converter."""
    include_page_breaks: bool = True
    include_frontmatter: bool = True
    max_pages: Optional[int] = None  # None = all pages
    page_separator: str = "\n\n---\n*Page {page}*\n\n"

class PDFConverter:
    """
    Extract text from PDF files.

    Uses pypdf for text extraction. Best effort - complex layouts
    may not be preserved perfectly.
    """

    def __init__(self, config: Optional[PDFConverterConfig] = None):
        self.config = config or PDFConverterConfig()

    def _parse_pdf_date(self, date_str: Optional[str]) -> Optional[str]:
        """
        Parse PDF date format to ISO format.

        PDF dates: D:YYYYMMDDHHmmSS or similar
        """
        if not date_str:
            return None

        # Remove D: prefix if present
        if date_str.startswith("D:"):
            date_str = date_str[2:]

        # Try to parse common formats
        try:
            # Basic format: YYYYMMDD
            if len(date_str) >= 8:
                year = date_str[0:4]
                month = date_str[4:6]
                day = date_str[6:8]
                return f"{year}-{month}-{day}"
        except (ValueError, IndexError):
            pass

        return date_str  # Return as-is if can't parse

    def _extract_metadata(self, reader: PdfReader) -> PDFMetadata:
        """Extract metadata from PDF."""
        meta = reader.metadata or {}

        return PDFMetadata(
            title=meta.get("/Title"),
            author=meta.get("/Author"),
            subject=meta.get("/Subject"),
            creator=meta.get("/Creator"),
            producer=meta.get("/Producer"),
            creation_date=self._parse_pdf_date(meta.get("/CreationDate")),
            modification_date=self._parse_pdf_date(meta.get("/ModDate")),
            pages=len(reader.pages),
        )

    def _clean_text(self, text: str) -> str:
        """Clean extracted text."""
        # Remove excessive whitespace
        text = re.sub(r"[ \t]+", " ", text)

        # Normalize line endings
        text = text.replace("\r\n", "\n").replace("\r", "\n")

        # Remove excessive blank lines
        text = re.sub(r"\n{4,}", "\n\n\n", text)

        # Strip lines
        lines = [line.strip() for line in text.split("\n")]
        text = "\n".join(lines)

        return text.strip()

    def _text_to_markdown(self, text: str) -> str:
        """
        Convert plain text to basic Markdown.

        - Detect likely headings (ALL CAPS lines, short lines followed by blank)
        - Preserve paragraph breaks
        """
        lines = text.split("\n")
        result_lines = []

        for i, line in enumerate(lines):
            stripped = line.strip()

            if not stripped:
                result_lines.append("")
                continue

            # Detect likely headings
            # - All caps, relatively short
            # - Followed by blank line or content
            is_heading = (
                stripped.isupper() and
                len(stripped) < 100 and
                len(stripped.split()) < 10 and
                not stripped.endswith(".")
            )

            if is_heading:
                # Determine heading level
                if i == 0 or (i > 0 and not lines[i-1].strip()):
                    result_lines.append(f"## {stripped.title()}")
                else:
                    result_lines.append(f"### {stripped.title()}")
            else:
                result_lines.append(stripped)

        return "\n".join(result_lines)

    def _compute_checksum(self, content: str) -> str:
        """Compute checksum of content."""
        return hashlib.sha256(content.encode()).hexdigest()[:16]

    def _generate_frontmatter(
        self,
        pdf_metadata: PDFMetadata,
        extra_metadata: dict,
    ) -> str:
        """Generate YAML frontmatter."""
        frontmatter = {
            "source": extra_metadata.get("source", "unknown"),
            "url": extra_metadata.get("url", ""),
            "title": pdf_metadata.title or extra_metadata.get("filename", "Untitled"),
            "filename": extra_metadata.get("filename", ""),
            "pages": pdf_metadata.pages,
            "crawled_at": datetime.now(timezone.utc).isoformat(),
        }

        # Add optional PDF metadata
        if pdf_metadata.author:
            frontmatter["author"] = pdf_metadata.author
        if pdf_metadata.creation_date:
            frontmatter["created_at"] = pdf_metadata.creation_date
        if pdf_metadata.modification_date:
            frontmatter["modified_at"] = pdf_metadata.modification_date

        # Add checksum
        if "checksum" in extra_metadata:
            frontmatter["checksum"] = extra_metadata["checksum"]

        yaml_str = yaml.dump(
            frontmatter,
            default_flow_style=False,
            allow_unicode=True,
            sort_keys=False,
        )

        return f"---\n{yaml_str}---\n\n"

    def extract_text(
        self,
        pdf_source: Union[bytes, Path, str],
    ) -> tuple[str, PDFMetadata]:
        """
        Extract raw text from PDF.

        Args:
            pdf_source: PDF bytes, file path, or path string

        Returns:
            Tuple of (extracted_text, pdf_metadata)
        """
        # Handle different input types
        if isinstance(pdf_source, bytes):
            from io import BytesIO
            reader = PdfReader(BytesIO(pdf_source))
        else:
            reader = PdfReader(pdf_source)

        # Extract metadata
        metadata = self._extract_metadata(reader)

        # Extract text from pages
        text_parts = []
        max_pages = self.config.max_pages or len(reader.pages)

        for i, page in enumerate(reader.pages[:max_pages]):
            page_text = page.extract_text() or ""
            page_text = self._clean_text(page_text)

            if page_text:
                if self.config.include_page_breaks and i > 0:
                    separator = self.config.page_separator.format(page=i + 1)
                    text_parts.append(separator)
                text_parts.append(page_text)

        full_text = "\n\n".join(text_parts)
        return full_text, metadata

    def convert(
        self,
        pdf_source: Union[bytes, Path, str],
        metadata: Optional[dict] = None,
    ) -> PDFConversionResult:
        """
        Convert PDF to Markdown with frontmatter.

        Args:
            pdf_source: PDF bytes, file path, or path string
            metadata: Optional extra metadata for frontmatter

        Returns:
            PDFConversionResult
        """
        metadata = metadata or {}
        warnings = []

        try:
            # Extract text
            text, pdf_metadata = self.extract_text(pdf_source)

            if not text.strip():
                warnings.append("No text extracted from PDF - may be scanned/image-based")

            # Convert to basic markdown
            markdown = self._text_to_markdown(text)

            # Compute checksum
            metadata["checksum"] = self._compute_checksum(markdown)

            # Generate frontmatter
            frontmatter_str = ""
            if self.config.include_frontmatter:
                frontmatter_str = self._generate_frontmatter(pdf_metadata, metadata)

            # Add title as H1 if available
            title = pdf_metadata.title or metadata.get("filename", "")
            if title:
                title_line = f"# {title}\n\n"
            else:
                title_line = ""

            full_markdown = frontmatter_str + title_line + markdown

            return PDFConversionResult(
                text=text,
                markdown=full_markdown,
                metadata=pdf_metadata,
                frontmatter=metadata,
                success=True,
                warnings=warnings,
            )

        except Exception as e:
            return PDFConversionResult(
                text="",
                markdown="",
                metadata=PDFMetadata(),
                frontmatter=metadata,
                success=False,
                error=str(e),
                warnings=warnings,
            )


def convert_pdf_to_markdown(
    pdf_source: Union[bytes, Path, str],
    metadata: Optional[dict] = None,
    config: Optional[PDFConverterConfig] = None,
) -> PDFConversionResult:
    """
    Convenience function to convert PDF to Markdown.

    Args:
        pdf_source: PDF bytes, file path, or path string
        metadata: Optional extra metadata
        config: Optional converter config

    Returns:
        PDFConversionResult
    """
    converter = PDFConverter(config)
    return converter.convert(pdf_source, metadata)
```

- Complex layouts may not be preserved
- Tables are extracted as plain text
