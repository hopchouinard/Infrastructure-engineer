```python
"""
Guides adapter for downloading PDF documentation from dl.ubnt.com.
"""

from dataclasses import dataclass
from typing import AsyncIterator, Optional
from urllib.parse import urljoin, urlparse, unquote
from datetime import datetime
import re

from bs4 import BeautifulSoup

from .base import BaseAdapter, CrawlItem, CrawlResult
from ..fetcher import Fetcher
from ..config import GuidesConfig

@dataclass
class GuideFile:
    """A guide file (PDF) from the guides directory."""
    name: str
    url: str
    size: Optional[str] = None  # e.g., "2.5M"
    size_bytes: Optional[int] = None
    last_modified: Optional[str] = None
    file_type: str = ".pdf"
```

### 4.3 Adapter Implementation

```python
class GuidesAdapter(BaseAdapter):
    """
    Adapter for downloading UniFi guides (PDFs).

    Discovers files by:
    1. Fetching directory listing page
    2. Parsing HTML table/links for file entries
    3. Filtering by configured file types
    """

    name = "guides"

    def __init__(self, config: GuidesConfig, fetcher: Fetcher):
        self.config = config
        self.fetcher = fetcher
        self.base_url = config.base_url.rstrip("/")

    def _parse_size(self, size_str: str) -> Optional[int]:
        """
        Parse size string to bytes.

        Examples: "2.5M" -> 2621440, "15K" -> 15360
        """
        if not size_str or size_str == "-":
            return None

        size_str = size_str.strip().upper()
        match = re.match(r"([\d.]+)([KMGT]?)", size_str)
        if not match:
            return None

        value = float(match.group(1))
        unit = match.group(2)

        multipliers = {
            "": 1,
            "K": 1024,
            "M": 1024 ** 2,
            "G": 1024 ** 3,
            "T": 1024 ** 4,
        }

        return int(value * multipliers.get(unit, 1))

    def _parse_date(self, date_str: str) -> Optional[str]:
        """Parse date string to ISO format."""
        if not date_str or date_str == "-":
            return None

        # Common formats: "2024-01-15 10:30", "15-Jan-2024 10:30"
        date_str = date_str.strip()

        # Try ISO-like format first
        if re.match(r"\d{4}-\d{2}-\d{2}", date_str):
            return date_str.split()[0] if " " in date_str else date_str

        # Try other common formats
        for fmt in ["%d-%b-%Y %H:%M", "%d-%b-%Y", "%Y-%m-%d %H:%M"]:
            try:
                dt = datetime.strptime(date_str, fmt)
                return dt.strftime("%Y-%m-%d")
            except ValueError:
                continue

        return date_str  # Return as-is if can't parse

    def _is_valid_file(self, filename: str) -> bool:
        """Check if file matches configured file types."""
        filename_lower = filename.lower()
        return any(
            filename_lower.endswith(ext.lower())
            for ext in self.config.file_types
        )

    async def _discover_files_from_listing(
        self,
        url: str,
    ) -> AsyncIterator[GuideFile]:
        """Parse directory listing and discover files."""
        text, result = await self.fetcher.get_text(url)

        if not result.success:
            return

        soup = BeautifulSoup(text, "html.parser")

        # Method 1: Parse table rows (Apache-style)
        rows = soup.select("table tr, tbody tr")
        for row in rows:
            cells = row.find_all(["td", "th"])
            if len(cells) >= 1:
                link = cells[0].find("a")
                if link and link.get("href"):
                    href = link["href"]
                    filename = unquote(href.split("/")[-1])

                    # Skip parent directory and non-matching files
                    if href == "../" or not self._is_valid_file(filename):
                        continue

                    # Extract metadata from other cells
                    size_str = cells[2].get_text(strip=True) if len(cells) > 2 else None
                    date_str = cells[1].get_text(strip=True) if len(cells) > 1 else None

                    file_url = urljoin(url + "/", href)

                    yield GuideFile(
                        name=filename,
                        url=file_url,
                        size=size_str,
                        size_bytes=self._parse_size(size_str) if size_str else None,
                        last_modified=self._parse_date(date_str) if date_str else None,
                        file_type=self._get_extension(filename),
                    )

        # Method 2: Parse plain links (simple listing)
        if not rows:
            links = soup.find_all("a")
            for link in links:
                href = link.get("href")
                if href:
                    filename = unquote(href.split("/")[-1])
                    if self._is_valid_file(filename):
                        file_url = urljoin(url + "/", href)
                        yield GuideFile(
                            name=filename,
                            url=file_url,
                            file_type=self._get_extension(filename),
                        )

    def _get_extension(self, filename: str) -> str:
        """Get file extension including the dot."""
        if "." in filename:
            return "." + filename.rsplit(".", 1)[-1].lower()
        return ""

    async def discover(self) -> AsyncIterator[CrawlItem]:
        """
        Discover all files to download from the guides directory.

        Yields CrawlItem for each file found.
        """
        async for guide_file in self._discover_files_from_listing(self.base_url):
            yield CrawlItem(
                url=guide_file.url,
                item_type="pdf",
                metadata={
                    "filename": guide_file.name,
                    "size": guide_file.size,
                    "size_bytes": guide_file.size_bytes,
                    "last_modified": guide_file.last_modified,
                    "file_type": guide_file.file_type,
                    "source": self.name,
                },
            )

    async def fetch(self, item: CrawlItem) -> CrawlResult:
        """
        Download a file.

        Returns CrawlResult with binary content.
        """
        result = await self.fetcher.get(item.url)

        if not result.success:
            return CrawlResult(
                url=item.url,
                content="",
                content_type=result.content_type,
                metadata=item.metadata,
                success=False,
                error=result.error,
            )

        # For binary files, content is bytes
        return CrawlResult(
            url=item.url,
            content=result.content,  # bytes
            content_type=result.content_type,
            metadata={
                **item.metadata,
                "actual_size_bytes": len(result.content),
                "headers": result.headers,
            },
            success=True,
        )

    async def get_file_info(self, url: str) -> Optional[GuideFile]:
        """
        Get file information using HEAD request.

        Useful for checking if file changed without downloading.
        """
        result = await self.fetcher.head(url)

        if not result.success:
            return None

        filename = unquote(urlparse(url).path.split("/")[-1])
        size_str = result.headers.get("content-length")
        last_mod = result.headers.get("last-modified")

        return GuideFile(
            name=filename,
            url=url,
            size=size_str,
            size_bytes=int(size_str) if size_str else None,
            last_modified=last_mod,
            file_type=self._get_extension(filename),
        )
```
