### 4.3 Configuration System

**config.py requirements:**

1. Load configuration from YAML file (optional)
2. Override with environment variables
3. Provide sensible defaults
4. Use Pydantic for validation
5. Raise `ConfigError` for invalid YAML or configuration values
6. Use `HttpUrl` for URL validation
7. Apply basic constraints (positive rates, valid depth ranges)
8. Export public API via `__all__`

```python
# src/doc_crawler/config.py
from pathlib import Path
from typing import Annotated

from pydantic import BaseModel, Field, HttpUrl
from pydantic_settings import BaseSettings, SettingsConfigDict

__all__ = [
    "ConfigError",
    "CrawlerConfig",
    "GuidesConfig",
    "HelpCenterConfig",
    "RateLimitConfig",
    "SourcesConfig",
]


class ConfigError(Exception):
    """Configuration loading or validation error."""
    pass


class RateLimitConfig(BaseModel):
    """Rate limiting configuration."""
    requests_per_second: Annotated[float, Field(gt=0, le=100)] = 1.0
    respect_robots_txt: bool = True


class HelpCenterConfig(BaseModel):
    """Help Center source configuration."""
    enabled: bool = True
    base_url: HttpUrl = HttpUrl("https://help.ui.com/hc/en-us")
    categories: list[str] = Field(default_factory=lambda: ["200320654-UniFi"])
    max_depth: Annotated[int, Field(ge=1, le=10)] = 3


class GuidesConfig(BaseModel):
    """Guides source configuration."""
    enabled: bool = True
    base_url: HttpUrl = HttpUrl("https://dl.ubnt.com/guides/UniFi/")
    file_types: list[str] = Field(default_factory=lambda: [".pdf"])


class SourcesConfig(BaseModel):
    """All documentation sources."""
    help_center: HelpCenterConfig = Field(default_factory=HelpCenterConfig)
    guides: GuidesConfig = Field(default_factory=GuidesConfig)


class CrawlerConfig(BaseSettings):
    """Main crawler configuration."""

    model_config = SettingsConfigDict(
        env_prefix="CRAWLER_",
        env_nested_delimiter="__",
    )

    # Output
    output_dir: Path = Path("ai_docs/vendor/unifi")

    # Rate limiting
    rate_limit: RateLimitConfig = Field(default_factory=RateLimitConfig)

    # Sources
    sources: SourcesConfig = Field(default_factory=SourcesConfig)

    # State
    state_file: str = "crawl-state.json"
    state_save_interval: Annotated[int, Field(ge=1)] = 50

    @classmethod
    def from_yaml(cls, path: Path) -> "CrawlerConfig":
        """Load configuration from YAML file.

        Raises:
            ConfigError: If YAML is invalid or configuration values are invalid.
        """
        import yaml

        if not path.exists():
            return cls()

        try:
            with open(path) as f:
                data = yaml.safe_load(f) or {}
        except yaml.YAMLError as e:
            raise ConfigError(f"Invalid YAML in {path}: {e}") from e

        try:
            return cls(**data)
        except Exception as e:
            raise ConfigError(f"Invalid configuration in {path}: {e}") from e
```

### 4.5 Base Adapter Interface

**adapters/base.py:**

The base adapter module defines immutable data classes and an abstract base class for adapters.

```python
from abc import ABC, abstractmethod
from collections.abc import AsyncIterator
from dataclasses import dataclass
from typing import Any

__all__ = [
    "BaseAdapter",
    "CrawlItem",
    "CrawlResult",
]


@dataclass(frozen=True)
class CrawlItem:
    """Represents a single item to be crawled.

    Note: Frozen dataclass for immutability. Use with_metadata() factory
    for dict-based metadata, get_metadata() to retrieve as dict.
    """
    url: str
    item_type: str  # "article", "pdf", "page"
    metadata: tuple[tuple[str, Any], ...] = ()

    @classmethod
    def with_metadata(
        cls, url: str, item_type: str, metadata: dict[str, Any] | None = None
    ) -> "CrawlItem":
        """Create a CrawlItem with a metadata dictionary."""
        meta_tuple = tuple(metadata.items()) if metadata else ()
        return cls(url=url, item_type=item_type, metadata=meta_tuple)

    def get_metadata(self) -> dict[str, Any]:
        """Get metadata as a dictionary."""
        return dict(self.metadata)


@dataclass(frozen=True)
class CrawlResult:
    """Result of crawling a single item.

    Note: Frozen dataclass for immutability. Use success_result() or
    failure_result() factory methods for convenience.
    """
    url: str
    success: bool
    content: str | bytes | None = None
    content_type: str | None = None
    metadata: tuple[tuple[str, Any], ...] = ()
    error: str | None = None

    @classmethod
    def success_result(
        cls,
        url: str,
        content: str | bytes,
        content_type: str,
        metadata: dict[str, Any] | None = None,
    ) -> "CrawlResult":
        """Create a successful crawl result."""
        meta_tuple = tuple(metadata.items()) if metadata else ()
        return cls(
            url=url, success=True, content=content,
            content_type=content_type, metadata=meta_tuple,
        )

    @classmethod
    def failure_result(
        cls, url: str, error: str, metadata: dict[str, Any] | None = None,
    ) -> "CrawlResult":
        """Create a failed crawl result."""
        meta_tuple = tuple(metadata.items()) if metadata else ()
        return cls(url=url, success=False, error=error, metadata=meta_tuple)

    def get_metadata(self) -> dict[str, Any]:
        """Get metadata as a dictionary."""
        return dict(self.metadata)


class BaseAdapter(ABC):
    """Base class for documentation source adapters.

    Subclasses must implement the name property, discover(), and fetch() methods.
    """

    @property
    @abstractmethod
    def name(self) -> str:
        """Unique identifier for this adapter."""
        ...

    @abstractmethod
    async def discover(self) -> AsyncIterator[CrawlItem]:
        """Discover all items to crawl from this source."""
        ...

    @abstractmethod
    async def fetch(self, item: CrawlItem) -> CrawlResult:
        """Fetch a single item."""
        ...
```

## 9. Design Decisions

This section documents the key design decisions made during implementation and the reasoning behind each choice.

### 9.1 Frozen Dataclasses for CrawlItem and CrawlResult

**Decision:** Use `@dataclass(frozen=True)` for `CrawlItem` and `CrawlResult`.

**Why:**
- **Immutability prevents bugs**: Crawl items pass through multiple stages (discovery → fetching → conversion). Mutable objects could be accidentally modified, causing hard-to-debug issues.
- **Thread safety**: Immutable objects are inherently thread-safe, important for concurrent crawling.
- **Hashability**: Frozen dataclasses are hashable, allowing use in sets for deduplication.

**Trade-off:** Required changing `metadata: dict` to `metadata: tuple[tuple[str, Any], ...]` since dicts are mutable. Added convenience methods (`with_metadata()`, `get_metadata()`) to preserve dict-like ergonomics.

### 9.2 Tuple Metadata Instead of Dict

**Decision:** Store metadata as `tuple[tuple[str, Any], ...]` instead of `dict[str, Any]`.

**Why:**
- **Required for frozen dataclasses**: Dicts are mutable and cannot be default values in frozen dataclasses.
- **Preserves insertion order**: Tuples maintain order, unlike pre-3.7 dicts.
- **Explicit immutability**: Makes it clear that metadata cannot be modified after creation.

**Mitigation:** Factory methods accept `dict[str, Any]` for convenience, and `get_metadata()` returns a dict for easy access.

### 9.3 Abstract Property for Adapter Name

**Decision:** Make `BaseAdapter.name` an `@abstractmethod` property instead of a class attribute.

**Why:**
- **Enforces implementation**: Class attributes are not enforced by ABC. Subclasses could forget to define `name`, causing runtime errors.
- **Flexibility**: Properties allow computed names if needed in the future.
- **Explicit contract**: Makes it clear that every adapter must have a name.

**Original spec:** Had `name: str` as a class attribute, which would not raise errors if omitted.

### 9.4 ConfigError Exception

**Decision:** Add a custom `ConfigError` exception for configuration-related errors.

**Why:**
- **Clear error messages**: Users get "Invalid YAML in config.yaml: ..." instead of raw YAML parsing exceptions.
- **Catch-all handling**: Callers can catch `ConfigError` to handle all config issues uniformly.
- **Separation of concerns**: Distinguishes configuration errors from other exceptions.

**Original spec:** Did not handle YAML parsing errors, which would propagate raw `yaml.YAMLError` exceptions.

### 9.5 HttpUrl Type for URL Fields

**Decision:** Use Pydantic's `HttpUrl` type instead of `str` for `base_url` fields.

**Why:**
- **Automatic validation**: Invalid URLs are rejected at configuration load time, not at crawl time.
- **Type safety**: Makes it explicit that these fields must be valid HTTP(S) URLs.
- **Normalization**: Pydantic normalizes URLs (e.g., adds trailing slashes if needed).

**Trade-off:** Requires `str(config.base_url)` when a string is needed. Added note in documentation.

### 9.6 Validation Constraints

**Decision:** Add Pydantic field constraints for configuration values.

**Why:**
- **Fail fast**: Invalid values (negative rate, zero depth) are caught at load time, not runtime.
- **Self-documenting**: Constraints in the schema document valid ranges.
- **Reasonable limits**: Prevents accidental DoS (e.g., 1000 requests/second) or infinite loops (depth=0).

**Constraints applied:**
- `requests_per_second`: > 0 and ≤ 100 (prevents DoS, requires positive value)
- `max_depth`: ≥ 1 and ≤ 10 (prevents infinite crawl, requires at least one level)
- `state_save_interval`: ≥ 1 (must checkpoint at least once)

**Scope consideration:** The original spec listed "Complex configuration validation" as out of scope. These constraints are considered "basic" validation (sanity checks), not "complex" (business logic validation).

### 9.7 Version from importlib.metadata

**Decision:** Read package version from `importlib.metadata` instead of hardcoding in `__init__.py`.

**Why:**
- **Single source of truth**: Version is defined only in `pyproject.toml`, not duplicated.
- **No sync issues**: Cannot forget to update `__init__.py` when bumping version.
- **Standard practice**: Recommended approach for modern Python packages.

**Implementation:**
```python
from importlib.metadata import version
__version__ = version("unifi-doc-crawler")
```

### 9.8 PEP 561 py.typed Marker

**Decision:** Include `py.typed` marker file for PEP 561 compliance.

**Why:**
- **Type checker support**: Allows mypy and other tools to use the package's type hints.
- **Library best practice**: Expected for typed Python packages.
- **Zero cost**: Empty file with no maintenance burden.

### 9.9 __all__ Exports in All Modules

**Decision:** Define `__all__` in every module.

**Why:**
- **Explicit public API**: Documents what is intended for external use.
- **IDE support**: Improves autocomplete and import suggestions.
- **Prevents accidental exports**: Internal helpers are not exposed.

## 10. Change Requests

This section documents modifications to the original task specification that were approved during implementation.

### CR-001: Add ConfigError Exception

**Original Spec:** `from_yaml()` did not handle errors:
```python
with open(path) as f:
    data = yaml.safe_load(f) or {}
return cls(**data)
```

**Change:** Added `ConfigError` exception and error handling.

**Reason:** Code review identified that YAML parsing errors would propagate as raw exceptions, confusing users.

**Approved:** Yes, as part of code review fixes.

### CR-002: Add Validation Constraints

**Original Spec:** No field constraints:
```python
requests_per_second: float = 1.0
max_depth: int = 3
```

**Change:** Added Pydantic field constraints.

**Reason:** Code review identified that invalid values (0, negative) would cause runtime errors or infinite loops.

**Approved:** Yes, classified as "basic validation" not "complex validation" (which remains out of scope).

### CR-003: Change base_url from str to HttpUrl

**Original Spec:** `base_url: str`

**Change:** `base_url: HttpUrl`

**Reason:** Provides automatic URL validation, catching typos early.

**Approved:** Yes, as part of code review fixes.

### CR-004: Make BaseAdapter.name an Abstract Property

**Original Spec:** `name: str` as class attribute.

**Change:** `@property @abstractmethod def name(self) -> str`

**Reason:** Class attributes are not enforced by ABC; subclasses could omit them.

**Approved:** Yes, as part of code review fixes.

### CR-005: Make CrawlItem and CrawlResult Frozen

**Original Spec:** Regular dataclasses with mutable dict metadata.

**Change:** Frozen dataclasses with tuple metadata and factory methods.

**Reason:** Immutability prevents bugs in concurrent crawling pipeline.

**Approved:** Yes, as part of code review fixes.

### CR-006: Make CrawlResult.content Optional

**Original Spec:** `content: str | bytes` (required).

**Change:** `content: str | bytes | None = None`

**Reason:** On failure, content has no meaningful value; forcing empty string is misleading.

**Approved:** Yes, as part of code review fixes.

### CR-007: Add py.typed Marker

**Original Spec:** Not mentioned.

**Change:** Added `src/doc_crawler/py.typed` for PEP 561 compliance.

**Reason:** Standard practice for typed packages, enables type checker support.

**Approved:** Yes, as part of code review fixes.

### CR-008: Add __all__ to All Modules

**Original Spec:** Not required.

**Change:** Added `__all__` exports to all modules.

**Reason:** Documents public API explicitly, improves IDE support.

**Approved:** Yes, as part of code review fixes.
